getgenv().gethui = function() return game.CoreGui end

local Rayfield = loadstring(game:HttpGet('https://sirius.menu/rayfield'))()

local Window = Rayfield:CreateWindow({
    Name = "TP + Poop + Tractor",
    LoadingTitle = "Loading...",
    LoadingSubtitle = "by RobloxPlayer",
    ConfigurationSaving = {Enabled = false},
    KeySystem = false,
})

local Tab = Window:CreateTab("Main",4483362458)

local Players = game:GetService("Players")
local Workspace = game:GetService("Workspace")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local LocalPlayer = Players.LocalPlayer

local SpawnToy = ReplicatedStorage.MenuToys.SpawnToyRemoteFunction
local DestroyToy = ReplicatedStorage.MenuToys.DestroyToy

local Vec3New = Vector3.new

local SCRIPT_ENABLED = false

-- Anti Network
local DROPPER_ENABLED = false

local DROP_HEIGHT = 8
local DROP_VELOCITY = Vec3New(0,20,0)
local HOLD_DROP_DELAY = 0.008

local function getCharacter()
	local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
	return char, char:WaitForChild("HumanoidRootPart")
end

local function getSpawnCFrame(hrp)
	return hrp.CFrame * CFrame.new(0,2,-3)
end

local function getToyFolder()
	for _,f in ipairs(Workspace:GetChildren()) do
		if f:IsA("Folder") and f.Name:find(LocalPlayer.Name) and f.Name:find("SpawnedInToys") then
			return f
		end
	end
end

-- 💩 spawn poop
local function spawnPoop()

	local char,hrp = getCharacter()

	task.defer(SpawnToy.InvokeServer,SpawnToy,"PoopPile",getSpawnCFrame(hrp),Vec3New(0,-16.157,0))

	local folder
	for i=1,30 do
		folder = getToyFolder()
		if folder then break end
		task.wait(0.05)
	end

	if not folder then return end

	for _,obj in ipairs(folder:GetChildren()) do
		if obj.Name == "PoopPile" then
			return obj
		end
	end
end

-- 💩 Anti Network Loop
local function PoopDropper()

	while DROPPER_ENABLED do

		local char,hrp = getCharacter()
		local poop = spawnPoop()

		if not poop then
			task.wait(0.2)
			continue
		end

		local holdRemote = poop.HoldPart:WaitForChild("HoldItemRemoteFunction")
		local dropRemote = poop.HoldPart:WaitForChild("DropItemRemoteFunction")

		while DROPPER_ENABLED and poop.Parent do

			local isHeld = poop:IsDescendantOf(char)

			if not isHeld then
				pcall(function()
					holdRemote:InvokeServer(poop,char)
				end)
			end

			pcall(function()

				local dropPos = hrp.CFrame * CFrame.new(0,DROP_HEIGHT,0)
				dropRemote:InvokeServer(poop,dropPos,DROP_VELOCITY)

			end)

			task.wait(HOLD_DROP_DELAY)

		end

		if poop and poop.Parent then
			pcall(function()
				DestroyToy:FireServer(poop)
			end)
		end

	end

end

-- 💩 normal poop loop
local function PoopLoop()

	while SCRIPT_ENABLED do
		local char,hrp = getCharacter()
		local poop = spawnPoop()

		if not poop then
			task.wait(0.2)
			continue
		end

		local holdRemote = poop.HoldPart:WaitForChild("HoldItemRemoteFunction")
		local dropRemote = poop.HoldPart:WaitForChild("DropItemRemoteFunction")

		while SCRIPT_ENABLED and poop.Parent do

			local isHeld = poop:IsDescendantOf(char)

			if not isHeld then
				pcall(function()
					holdRemote:InvokeServer(poop,char)
				end)
			end

			pcall(function()
				local dropPos = hrp.CFrame * CFrame.new(0,DROP_HEIGHT,0)
				dropRemote:InvokeServer(poop,dropPos,DROP_VELOCITY)
			end)

			task.wait(HOLD_DROP_DELAY)

		end

		if poop and poop.Parent then
			pcall(function()
				DestroyToy:FireServer(poop)
			end)
		end

	end
end

-- 🚜 tractor spawn
local function SpawnTractor()

	local char,hrp = getCharacter()
	local hum = char:WaitForChild("Humanoid")

	local folder = getToyFolder()

	if folder then
		for _,v in ipairs(folder:GetChildren()) do
			if v.Name == "TractorGreen" then
				DestroyToy:FireServer(v)
			end
		end
	end

	task.wait(0.5)

	SpawnToy:InvokeServer(
		"TractorGreen",
		hrp.CFrame * CFrame.new(0,0,-6),
		Vector3.new(0,-16.157,0)
	)

	for i=1,50 do
		folder = getToyFolder()
		if folder then break end
		task.wait(0.05)
	end

	if not folder then return end

	local tractor

	for _,v in ipairs(folder:GetChildren()) do
		if v.Name == "TractorGreen" then
			tractor = v
			break
		end
	end

	if not tractor then return end

	local seat = tractor:FindFirstChildWhichIsA("VehicleSeat",true)

	if seat then
		seat:Sit(hum)
	end

end

-- ⭐ メイン処理
local function StartAll()

	task.spawn(PoopLoop)

	local char,hrp = getCharacter()
	local startPos = hrp.Position

	local offsets = {
		Vector3.new(-1.18,46.55,-3.84),
		Vector3.new(-75.08,24.19,61.86),
		Vector3.new(5.36,89.01,20.19),
		Vector3.new(70.57,27.01,35.63),
		Vector3.new(-40.22,60.11,18.45),
		Vector3.new(15.73,75.90,-42.66),
		Vector3.new(88.31,33.27,-10.58),
		Vector3.new(-62.49,52.14,73.92),
		Vector3.new(24.80,95.66,12.77)
	}

	task.spawn(function()
		while SCRIPT_ENABLED do
			for _,offset in ipairs(offsets) do
				if not SCRIPT_ENABLED then break end
				if hrp.Parent then
					hrp.CFrame = CFrame.new(startPos + offset)
				end
				task.wait(0.05)
			end
		end
	end)

	task.spawn(function()
		task.wait(2)
		if SCRIPT_ENABLED then
			SpawnTractor()
		end
	end)

end

-- 🔁 リスポーン対応
LocalPlayer.CharacterAdded:Connect(function()
	if SCRIPT_ENABLED then
		StartAll()
	end
end)

-- 🔘 Start Everything
Tab:CreateToggle({
	Name = "Start Everything",
	CurrentValue = false,
	Callback = function(Value)

		SCRIPT_ENABLED = Value

		if Value then
			StartAll()
		end

	end,
})

-- 🔘 Anti Network
Tab:CreateToggle({
	Name = "Anti network",
	CurrentValue = false,
	Callback = function(Value)

		DROPPER_ENABLED = Value

		if Value then
			task.spawn(PoopDropper)
		end

	end,
})

Rayfield:LoadConfiguration()
