    -- Anti-Cheat Metatable Bypass (State Spoofer)
    local MetatableBypassEnabled = false
    local spoofedProps = {
        WalkSpeed = {},
        JumpPower = {},
        JumpHeight = {}
    }
    
    local function initMetatableBypass()
        local rawMT = getrawmetatable and getrawmetatable(game)
        if not rawMT then return end
        
        local oldIndex = rawMT.__index
        local oldNewIndex = rawMT.__newindex
        local setReadOnly = setreadonly or make_writeable
        
        if not oldIndex or not oldNewIndex or not setReadOnly then return end
        
        setReadOnly(rawMT, false)
        
        rawMT.__index = newcclosure(function(self, key)
            if MetatableBypassEnabled and not checkcaller() and typeof(self) == "Instance" then
                if self:IsA("Humanoid") then
                    if key == "WalkSpeed" then
                        return spoofedProps.WalkSpeed[self] or 16
                    elseif key == "JumpPower" then
                        return spoofedProps.JumpPower[self] or 50
                    elseif key == "JumpHeight" then
                        return spoofedProps.JumpHeight[self] or 7.2
                    end
                elseif self:IsA("BasePart") and (self.Name == "HumanoidRootPart" or self.Name == "Torso") then
                    if key == "Velocity" or key == "AssemblyLinearVelocity" then
                        return Vector3.new(0, 0, 0)
                    end
                end
            end
            return oldIndex(self, key)
        end)
        
        rawMT.__newindex = newcclosure(function(self, key, value)
            if MetatableBypassEnabled and not checkcaller() and typeof(self) == "Instance" and self:IsA("Humanoid") then
                if key == "WalkSpeed" then
                    spoofedProps.WalkSpeed[self] = value
                elseif key == "JumpPower" then
                    spoofedProps.JumpPower[self] = value
                elseif key == "JumpHeight" then
                    spoofedProps.JumpHeight[self] = value
                end
            end
            return oldNewIndex(self, key, value)
        end)
        
        setReadOnly(rawMT, true)
    end
    
    pcall(initMetatableBypass)

    local function spawnBlobman()
        local args = {"CreatureBlobman", CFrame.new(0, 5000000, 0), Vector3.new(0, 60, 0)}
        pcall(function()
            Services.RS.MenuToys.SpawnToyRemoteFunction:InvokeServer(unpack(args))

        if folder and folder:FindFirstChild("CreatureBlobman") then
            blobman = folder.CreatureBlobman

            end
        end

    local function startAntiGucci()
        local character = Services.Player.Character or Services.Player.CharacterAdded:Wait()
        local humanoid = character:WaitForChild("Humanoid")
        local rootPart = character:WaitForChild("HumanoidRootPart")
        safePositionBlob = rootPart.Position

        local folder = workspace:FindFirstChild(Services.Player.Name .. "SpawnedInToys")
        local blob = folder and folder:FindFirstChild("CreatureBlobman")
        local seat = blob and blob:FindFirstChild("VehicleSeat")

        if seat and seat:IsA("VehicleSeat") then
            rootPart.CFrame = seat.CFrame + Vector3.new(0, 2, 0)
            seat:Sit(humanoid)
        end

        humanoid:GetPropertyChangedSignal("Jump"):Connect(function()
            if humanoid.Jump and humanoid.Sit then
                restoreFramesBlob = 15
                safePositionBlob = rootPart.Position
            end
        end)

        if antiGucciConnection then antiGucciConnection:Disconnect() end
        antiGucciConnection = Services.RunService.Heartbeat:Connect(function()
            if not rootPart or not humanoid then return end
            pcall(function()
                Services.RS.CharacterEvents.RagdollRemote:FireServer(rootPart, 0)
            end)
            if restoreFramesBlob > 0 then
                rootPart.CFrame = CFrame.new(safePositionBlob)
                restoreFramesBlob = 1
            end
        end)

        task.spawn(function()
            while humanoid.Sit do task.wait(1) end
            task.wait(0.5)
            rootPart.CFrame = CFrame.new(safePositionBlob)
        end)
    end

    local function stopAntiGucci()
        if antiGucciConnection then antiGucciConnection:Disconnect() antiGucciConnection = nil end
        if blobman then
            pcall(function()
                Services.RS.MenuToys.DestroyToy:FireServer(blobman)
            end)
            blobman = nil
        end
    end

    local tractorSystemEnabled = false
    local tractor = nil
    local seat = nil
    local tractorRespawnConnection = nil
    local tractorWatcherConnection = nil
    local tractorDeathConnection = nil
    local ragdollConnection = nil
    local seatGuardConnection = nil

    local function isValidTractor(t)
        return t and t.Parent and t:IsDescendantOf(workspace) and (t:FindFirstChildWhichIsA("VehicleSeat", true) or t:FindFirstChild("VehicleSeat", true))
    end

    local function getTractorAndSeat()
        local folder = workspace:FindFirstChild(Services.Player.Name .. "SpawnedInToys")
        local function scanParent(parent)
            if not parent then return nil, nil end
            for _, child in ipairs(parent:GetChildren()) do
                if child:IsA("Model") and child.Name == "TractorGreen" then
                    local s = child:FindFirstChildWhichIsA("VehicleSeat", true) or child:FindFirstChild("VehicleSeat", true)
                    if s then
                        return child, s
                    end
                end
            end
            return nil, nil
        end
        if folder then
            local t, s = scanParent(folder)
            if t and s then return t, s end
        end
        return scanParent(workspace)
    end

    local function spawnTractor()
        local baseCFrame = CFrame.new(0, 10000000000, 0) * CFrame.Angles(-2.5694754123687744, 0.10936412960290909, 3.0714223384857178)
        local SpawnRemote = Services.MenuToys and Services.MenuToys:FindFirstChild("SpawnToyRemoteFunction")
        if not SpawnRemote then return nil, nil end
        
        local foundTr, foundSeat = nil, nil
        local startSpawn = tick()
        repeat
            task.spawn(function()
                pcall(function()
                    SpawnRemote:InvokeServer("TractorGreen", baseCFrame, Vector3.new(0, -49.45, 0))
                end)
            end)
            task.wait(0.2)
            foundTr, foundSeat = getTractorAndSeat()
        until (foundTr and foundSeat) or (tick() - startSpawn > 5) or not tractorSystemEnabled
        
        return foundTr, foundSeat
    end

    local function anchorPulseForModel(model)
        if not model then return end
        task.spawn(function()
            for _, part in ipairs(model:GetDescendants()) do
                if part and part:IsA("BasePart") then
                    pcall(function()
                        part.Anchored = true
                        part.CanCollide = false
                        part.CanTouch = false
                        Services.RunService.Heartbeat:Wait()
                        part.Anchored = false
                    end)
                end
            end
        end)
    end

    local function destroySeatWelds(seatObj)
        if not seatObj or not seatObj.Parent then return end
        for _, child in pairs(seatObj:GetChildren()) do
            if child:IsA("Weld") or child:IsA("WeldConstraint") or child:IsA("Snap") then
                pcall(function()
                    child.Part0 = nil
                    child.Part1 = nil
                    child:Destroy()
                end)
            end
        end
        for _, desc in pairs(seatObj.Parent:GetDescendants()) do
            if (desc:IsA("Weld") or desc:IsA("WeldConstraint") or desc:IsA("Snap")) and (desc.Part0 == seatObj or desc.Part1 == seatObj) then
                pcall(function()
                    desc.Part0 = nil
                    desc.Part1 = nil
                    desc:Destroy()
                end)
            end
        end
    end

    local function gucci(seatObj)
        if not seatObj or not seatObj.Parent then return end
        local char = Utils.getCharacter()
        local hum = Utils.getHumanoid()
        local hrp = Utils.getHRP()
        if not (char and hum and hrp) then return end
        
        local startTime = tick()
        local loopConn
        loopConn = Services.RunService.Heartbeat:Connect(function()
            if not hrp.Parent or not tractorSystemEnabled or tick() - startTime >= 1.5 then
                loopConn:Disconnect()
                return
            end
            pcall(function()
                Services.RS.CharacterEvents.RagdollRemote:FireServer(hrp, -math.huge)
            end)
        end)
        
        seatObj.ProximityPrompt.Enabled = false
        pcall(function() hum.Sit = true end)
        pcall(function()
            Services.RS.CharacterEvents.RagdollRemote:FireServer(hrp, -math.huge)
        end)
        
        task.wait(0.09)
        
        pcall(function()
            if seatObj:FindFirstChild("WeldConstraint") then
                seatObj.WeldConstraint.Part1 = nil
            end
        end)
        destroySeatWelds(seatObj)
        task.wait(0.05)
        
        anchorPulseForModel(seatObj.Parent)
        task.wait(0.09)
        
        local attempts = 0
        repeat
            pcall(function()
                Services.RS.CharacterEvents.RagdollRemote:FireServer(hrp, -math.huge)
                anchorPulseForModel(seatObj.Parent)
                seatObj:Sit(hum)
                attempts = attempts + 1
                Services.RunService.Heartbeat:Wait()
            end)
        until (seatObj and seatObj.Occupant == hum) or attempts >= 50 or not tractorSystemEnabled
        
        pcall(function() loopConn:Disconnect() end)
        pcall(function()
            Services.RS.CharacterEvents.RagdollRemote:FireServer(hrp, -math.huge)
        end)
        
        pcall(function()
            if not seatObj or not seatObj.Parent then return end
            if seatObj.Occupant == hum then
                hum.Sit = false
                Services.RunService.Heartbeat:Wait()
                
                local possibleh = seatObj.Parent:FindFirstChild("HumanoidRootPart") or seatObj.Parent.PrimaryPart
                if possibleh and seatObj:FindFirstChild("WeldConstraint") then
                    seatObj.WeldConstraint.Part1 = possibleh
                end
                
                local bv = seatObj:FindFirstChild("BodyVelocity") or Instance.new("BodyVelocity")
                bv.Velocity = Vector3.new(0, 99999, 0)
                bv.MaxForce = Vector3.new(0, 9999999, 0)
                bv.P = 15000
                bv.Parent = seatObj
                
                Services.RS.CharacterEvents.RagdollRemote:FireServer(hrp, 0.9)
            end
        end)
    end

    local function startSeatWatcher()
        if seatGuardConnection then seatGuardConnection:Disconnect(); seatGuardConnection = nil end
        if not seat or not seat.Parent then return end
        
        local function handleOccupantChange()
            if not tractorSystemEnabled or not seat or not seat.Parent then return end
            local occupant = seat.Occupant
            local char = Utils.getCharacter()
            if occupant and occupant.Parent and occupant.Parent ~= char then
                pcall(function()
                    seat:Sit(nil)
                    occupant.Sit = false
                    occupant.PlatformStand = true
                    task.delay(0.1, function() if occupant then occupant.PlatformStand = false end end)
                end)
                pcall(function() gucci(seat) end)
            end
        end
        
        seatGuardConnection = seat:GetPropertyChangedSignal("Occupant"):Connect(handleOccupantChange)
        task.spawn(handleOccupantChange)
    end

    local function setupRespawnListener(currentTractor)
        if tractorRespawnConnection then tractorRespawnConnection:Disconnect(); tractorRespawnConnection = nil end
        if not currentTractor then return end
        
        seat = currentTractor:FindFirstChildWhichIsA("VehicleSeat", true) or currentTractor:FindFirstChild("VehicleSeat", true)
        startSeatWatcher()
        
        tractorRespawnConnection = currentTractor.AncestryChanged:Connect(function(_, parent)
            if not tractorSystemEnabled or parent or respawning then return end
            respawning = true
            task.wait(0.15)
            
            local newTr, newSeat = spawnTractor()
            if tractorSystemEnabled and newTr and newSeat and isValidTractor(newTr) then
                tractor = newTr
                seat = newSeat
                setupRespawnListener(tractor)
                pcall(function() gucci(seat) end)
            end
            respawning = false
        end)
    end

    local function connectDeathHandler()
        if tractorDeathConnection then tractorDeathConnection:Disconnect(); tractorDeathConnection = nil end
        local hum = Utils.getHumanoid()
        if not hum then return end
        
        tractorDeathConnection = hum.Died:Connect(function()
            if not tractorSystemEnabled then return end
            local oldTractor = tractor
            tractor = nil
            seat = nil
            respawning = true
            
            Services.Player.CharacterAdded:Wait()
            task.wait(0.5)
            
            pcall(function()
                local destroyrem = Services.MenuToys and Services.MenuToys:FindFirstChild("DestroyToy")
                if oldTractor and oldTractor.Parent and destroyrem then
                    destroyrem:FireServer(oldTractor)
                end
            end)
            
            local newTr, newSeat = spawnTractor()
            if tractorSystemEnabled and newTr and newSeat and isValidTractor(newTr) then
                tractor = newTr
                seat = newSeat
                respawning = false
                pcall(function() gucci(seat) end)
                setupRespawnListener(tractor)
            else
                respawning = false
            end
            connectDeathHandler()
        end)
    end

    local function startTractorWatcher()
        if tractorWatcherConnection then tractorWatcherConnection:Disconnect(); tractorWatcherConnection = nil end
        
        tractorWatcherConnection = Services.RunService.Heartbeat:Connect(function()
            if not tractorSystemEnabled or respawning then return end
            local tExists = isValidTractor(tractor)
            if not tExists then
                local newTr2, newSeat2 = getTractorAndSeat()
                if newTr2 and newSeat2 and isValidTractor(newTr2) then
                    tractor = newTr2
                    seat = newSeat2
                    setupRespawnListener(tractor)
                    pcall(function() gucci(seat) end)
                    return
                end
                
                respawning = true
                task.spawn(function()
                    task.wait(0.06)
                    local newT, newS = spawnTractor()
                    if tractorSystemEnabled and newT and newS and isValidTractor(newT) then
                        tractor = newT
                        seat = newS
                        setupRespawnListener(tractor)
                        pcall(function() gucci(seat) end)
                    else
                        tractor = nil
                        seat = nil
                    end
                    respawning = false
                end)
            else
                local seatValid = seat and seat.Parent and seat:IsDescendantOf(workspace)
                if not seatValid then
                    if tractor and tractor.Parent then
                        seat = tractor:FindFirstChildWhichIsA("VehicleSeat", true) or tractor:FindFirstChild("VehicleSeat", true)
                    end
                    if not seat or not seat.Parent then
                        tractor = nil

            truePart:Destroy()

                    setMassless()

                        if truePart and not truePart:FindFirstChild("RootAttachment") then

        end
    end
