return function(ctx)
    local Services = ctx.Services
    local Connect = ctx.ConnectionManager.Connect
    local Disconnect = ctx.ConnectionManager.Disconnect
    local Utils = ctx.Utils
    local Tabs = ctx.Tabs
    local Toggles = ctx.Toggles
    local Options = ctx.Options
    
    local TargetSelect = Tabs.Target:AddLeftGroupbox("Select Target")
    local TargetKick = Tabs.Target:AddLeftGroupbox("Kick Methods")
    local TargetUtils = Tabs.Target:AddRightGroupbox("Target Utils")
    local WhitelistGroup = Tabs.Target:AddRightGroupbox("Whitelist")
    
    ctx.selectedKickPlayer = nil
    
    TargetSelect:AddDropdown("KickPlayerDropdown", {
        Values = Utils.getPlayerList(),
        Default = 1,
        Multi = false,
        Text = "Select Player",
        Callback = function(Value)
            ctx.selectedKickPlayer = Utils.getPlayerFromSelection(Value)
        end,
    })
    
    TargetSelect:AddButton({
        Text = "Refresh Player List",
        Func = function()
            Options.KickPlayerDropdown:SetValues(Utils.getPlayerList())
            Options.KickPlayerDropdown:SetValue(nil)
            ctx.selectedKickPlayer = nil
        end
    })
    
    TargetSelect:AddButton({
        Text = "TP to Player",
        Func = function()
            if not ctx.selectedKickPlayer then
                Utils.notify("Error", "Select a target first", 3)
                return
            end
            local myHrp = Utils.getHRP()
            local tHrp = Utils.getHRP(ctx.selectedKickPlayer)
            if myHrp and tHrp then
                myHrp.CFrame = tHrp.CFrame * CFrame.new(0, 3, 0)
                Utils.notify("System", "Teleported to " .. ctx.selectedKickPlayer.Name, 3)
            end
        end
    })
    
    -- Kick Spam Grab
    TargetKick:AddToggle("KickSpamGrab", {
        Text = "Kick (spam grab)",
        Default = false,
        Callback = function(on)
            if not on then return end
            task.spawn(function()
                local GE = Services.GrabEvents
                if not GE then return end
                local myRoot = Utils.getHRP()
                if not myRoot then
                    Toggles.KickSpamGrab:SetValue(false)
                    return
                end
                local savedPos = myRoot.CFrame
                local dragging = false
                local grabStartTime = 0
                while Toggles.KickSpamGrab and Toggles.KickSpamGrab.Value do
                    local target = ctx.selectedKickPlayer
                    if not target or not target.Parent then break end
                    myRoot = Utils.getHRP()
                    if not myRoot then break end
                    local tRoot = Utils.getHRP(target)
                    local tHum = Utils.getHumanoid(target)
                    if tRoot and tHum and tHum.Health > 0 then
                        tRoot.AssemblyLinearVelocity = Vector3.zero
                        tRoot.AssemblyAngularVelocity = Vector3.zero
                        if not dragging then
                            myRoot.CFrame = tRoot.CFrame
                            myRoot.Velocity = Vector3.zero
                            pcall(function()
                                tHum.PlatformStand = true
                                tHum.Sit = true
                                GE.SetNetworkOwner:FireServer(tRoot, myRoot.CFrame)
                                GE.CreateGrabLine:FireServer(tRoot, Vector3.zero, tRoot.Position, false)
                            end)
                            if grabStartTime == 0 then grabStartTime = tick() end
                            if tick() - grabStartTime > 0.35 then
                                dragging = true
                                grabStartTime = 0
                            end
                        else
                            myRoot.CFrame = savedPos
                            myRoot.Velocity = Vector3.zero
                            local lockPos = savedPos * CFrame.new(0, 17, 0)
                            tRoot.CFrame = lockPos
                            tRoot.Velocity = Vector3.zero
                            pcall(function()
                                tHum.PlatformStand = true
                                tHum.Sit = false
                                GE.SetNetworkOwner:FireServer(tRoot, lockPos)
                                GE.DestroyGrabLine:FireServer(tRoot)
                                GE.CreateGrabLine:FireServer(tRoot, Vector3.zero, tRoot.Position, false)
                            end)
                        end
                    else
                        dragging = false
                        grabStartTime = 0
                    end
                    Services.RunService.Heartbeat:Wait()
                end
                if myRoot then
                    myRoot.CFrame = savedPos
                    myRoot.Velocity = Vector3.zero
                end
                Toggles.KickSpamGrab:SetValue(false)
            end)
        end
    })

    -- Try to Kick Target (Blobman)
    TargetKick:AddButton({
        Text = "Try to Kick Target (Blobman)",
        Func = function()
            local target = ctx.selectedKickPlayer
            if not target then
                Utils.notify("Error", "Select target first", 3)
                return
            end
            local char = Utils.getCharacter()
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            local seat = hum and hum.SeatPart
            if not seat or seat.Parent.Name ~= "CreatureBlobman" then
                Utils.notify("Error", "Sit on Blobman first", 3)
                return
            end
            task.spawn(function()
                local myChar = Utils.getCharacter()
                local tChar = target.Character
                if not myChar or not tChar then return end
                
                local myHRP = myChar:FindFirstChild("HumanoidRootPart")
                local tHRP = tChar:FindFirstChild("HumanoidRootPart")
                local tHead = tChar:FindFirstChild("Head")
                local tHum = tChar:FindFirstChildOfClass("Humanoid")
                
                if not myHRP or not tHRP or not tHead or not tHum then return end
                
                local myBlob = seat.Parent
                local leftDetector = myBlob:FindFirstChild("LeftDetector")
                if not leftDetector then return end
                local weld = leftDetector:FindFirstChild("LeftWeld") or leftDetector:FindFirstChildWhichIsA("Weld")
                if not weld then return end
                
                -- Clear existing BodyPositions on target HRP
                for _, v in pairs(tHRP:GetChildren()) do
                    if v.Name == "BodyPosition" then v:Destroy() end
                end
                
                local backUp = myHRP.CFrame
                local bp = Instance.new("BodyPosition")
                bp.Parent = tHRP
                bp.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
                bp.D = 105
                bp.P = 8000
                bp.Position = backUp.Position + Vector3.new(0, 12.5, 0)
                
                local function blob_kick_func(blob, hrp, rl, action)
                    local detec = blob:FindFirstChild(rl .. "Detector")
                    if detec then
                        local grabRemote = blob.BlobmanSeatAndOwnerScript.CreatureGrab
                        local dropRemote = blob.BlobmanSeatAndOwnerScript.CreatureDrop
                        local relRemote = blob.BlobmanSeatAndOwnerScript.CreatureRelease
                        local weldName = rl .. "Weld"
                        local weldConstraint = detec:FindFirstChild(weldName) or detec:FindFirstChildWhichIsA("Weld")
                        if action == "Default" then
                            pcall(function() grabRemote:FireServer(detec, hrp, weldConstraint) end)
                        elseif action == "DDrop" then
                            pcall(function() dropRemote:FireServer(weldConstraint) end)
                        elseif action == "Release" then
                            pcall(function() relRemote:FireServer(weldConstraint, hrp) end)
                        end
                    end
                end
                
                local tryProcess = true
                local startTime = tick()
                while weld.Attachment0 ~= tHRP.RootAttachment and tick() - startTime < 5 do
                    if hum.SeatPart == seat then
                        myHRP.CFrame = tHRP.CFrame - Vector3.new(0, 7.5, 0)
                        blob_kick_func(myBlob, tHRP, "Left", "Default")
                        task.wait()
                        blob_kick_func(myBlob, tHRP, "Left", "Release")
                    else
                        tryProcess = false
                        break
                    end
                end
                
                if tryProcess then
                    myHRP.CFrame = backUp
                    task.wait(0.15)
                    
                    local ownerTime = tick()
                    while not (tHead:FindFirstChild("PartOwner") and isnetworkowner(tHRP)) and hum.SeatPart == seat and tick() - ownerTime < 5 do
                        pcall(function()
                            Services.GrabEvents.CreateGrabLine:FireServer(tHRP, Vector3.zero, tHRP.Position, false)
                        end)
                        task.wait()
                    end
                    
                    pcall(function()
                        Services.GrabEvents.DestroyGrabLine:FireServer(tHRP)
                    end)
                    
                    task.delay(0.5, function() bp:Destroy() end)
                    
                    while tHRP.Parent and tHRP and tHum.Health > 0 and hum.SeatPart == seat do
                        blob_kick_func(myBlob, tHRP, "Left", "Default")
                        blob_kick_func(myBlob, tHRP, "Right", "Default")
                        task.wait()
                        blob_kick_func(myBlob, tHRP, "Left", "DDrop")
                        blob_kick_func(myBlob, tHRP, "Right", "DDrop")
                    end
                else
                    pcall(function() bp:Destroy() end)
                end
            end)
        end
    })

    -- Fling
    TargetKick:AddToggle("FlingToggle", {
        Text = "Fling",
        Default = false,
        Callback = function(on)
            if on then
                if not ctx.selectedKickPlayer then
                    Utils.notify("System", "Select target first!", 3)
                    Toggles.FlingToggle:SetValue(false)
                    return
                end
                local myHrp = Utils.getHRP()
                local originalPos = myHrp and myHrp.CFrame
                local flingBAV = nil
                task.spawn(function()
                    while Toggles.FlingToggle and Toggles.FlingToggle.Value do
                        local target = ctx.selectedKickPlayer
                        local hrp = Utils.getHRP()
                        local hum = Utils.getHumanoid()
                        if hrp and hum then
                            if target and target.Parent then
                                local tRoot = Utils.getHRP(target)
                                local tHum = Utils.getHumanoid(target)
                                if tRoot and tHum and tHum.Health > 0 then
                                    if not flingBAV or flingBAV.Parent ~= hrp then
                                        if flingBAV then flingBAV:Destroy() end
                                        flingBAV = Instance.new("BodyAngularVelocity")
                                        flingBAV.Name = "FlingSpin"
                                        flingBAV.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
                                        flingBAV.AngularVelocity = Vector3.new(0, 10000, 0)
                                        flingBAV.P = 10000
                                        flingBAV.Parent = hrp
                                    end
                                    for _, part in pairs(Utils.getCharacter():GetDescendants()) do
                                        if part:IsA("BasePart") then part.CanCollide = false end
                                    end
                                    local loop
                                    loop = Services.RunService.Heartbeat:Connect(function()
                                        if not (Toggles.FlingToggle and Toggles.FlingToggle.Value) or not tRoot or not tRoot.Parent then 
                                            if loop then loop:Disconnect() end
                                            return 
                                        end
                                        hrp.CFrame = tRoot.CFrame
                                        hrp.Velocity = Vector3.zero
                                    end)
                                    local startTime = tick()
                                    while tick() - startTime < 1.5 do
                                        if not (Toggles.FlingToggle and Toggles.FlingToggle.Value) or not tRoot.Parent then break end
                                        task.wait(0.1)
                                    end
                                    if loop then loop:Disconnect() end
                                else
                                    task.wait(0.2)
                                end
                            else
                                Toggles.FlingToggle:SetValue(false)
                            end
                            task.wait(0.1)
                        else
                            task.wait(0.5)
                        end
                    end
                    if flingBAV then flingBAV:Destroy() end
                    local char = Utils.getCharacter()
                    if char then
                        for _, part in pairs(char:GetDescendants()) do
                            if part:IsA("BasePart") then part.CanCollide = true end
                        end
                        local hrp = char:FindFirstChild("HumanoidRootPart")
                        if hrp then
                            hrp.RotVelocity = Vector3.zero
                            hrp.Velocity = Vector3.zero
                            if originalPos then hrp.CFrame = originalPos end
                        end
                    end
                end)
            end
        end
    })

                                hp.DropItemRemoteFunction:InvokeServer(b, CFrame.new(hrp.Position + Vector3.new(0, -2000, 0)), Vector3.zero)
