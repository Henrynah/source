local function getplot()
    for i = 1, 5 do
        local plot = workspace.Plots:FindFirstChild("Plot"..i)
        local value = plot.PlotSign.ThisPlotsOwners:FindFirstChild("Value")

    end
end

local function spawntoy(toy, cf)
    if not plr.CanSpawnToy.Value then plr.CanSpawnToy.Changed:Wait() end
    local t
    local toyadded = inv.ChildAdded:Connect(function(c)
        if c.Name == toy then t = c toyadded:Disconnect() end
    end)
    task.spawn(function() SpawnToy:InvokeServer(toy, cf, Vector3.new(0,0,0)) end)
    local time = tick() + 1
    repeat task.wait() until t or tick() > time
    return t or (getplot() and workspace.PlotItems[getplot().Name]:FindFirstChild(toy))
end

-- Blacklist persistente
local BlacklistFile = "Blacklist_Config.json"

local function SaveBlacklist()
    local data = {Blacklist = Blacklist or {}, Enabled = BlacklistEnabled or false}
    writefile(BlacklistFile, HttpService:JSONEncode(data))
end

local function LoadBlacklist()
    if isfile(BlacklistFile) then
        local success, data = pcall(function() return HttpService:JSONDecode(readfile(BlacklistFile)) end)
        if success then 
            Blacklist = data.Blacklist or {} 
            BlacklistEnabled = data.Enabled or false 
            return 
        end
    end
    Blacklist, BlacklistEnabled = {}, false
end
LoadBlacklist()
