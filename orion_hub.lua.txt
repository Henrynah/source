local function GetPlayerDisplayList()
    local opts = {}
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then
            table.insert(opts, p.DisplayName .. " (" .. p.Name .. ")")
        end
    end
    return opts
end

local function GetUserNameFromDisplay(displayString)
    if not displayString or displayString == "" then return nil end
    local startPos, endPos = string.find(displayString, "%(")
    if startPos then
        local userName = string.sub(displayString, startPos + 1, -2)
        return userName
    end
    return nil
end

-- ============================================
-- 全プレイヤー取得関数
-- ============================================
local function GetAllPlayers()
    local allPlayers = {}
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            table.insert(allPlayers, player)
        end
    end
    return allPlayers
end

-- IsWhitelisted を事前定義
local function IsWhitelisted(player)
    if not player then return false end
    return Whitelist[player.Name] == true
end

local function GetAllPlayersIncludingInPlot()
    local allPlayers = GetAllPlayers()
    local plotItems = Workspace:FindFirstChild("PlotItems")
    if plotItems then
        local playersInPlots = plotItems:FindFirstChild("PlayersInPlots")
        if playersInPlots then
            for _, playerObj in ipairs(playersInPlots:GetChildren()) do
                if playerObj:IsA("Folder") or playerObj:IsA("Model") then
                    local playerName = playerObj.Name
                    local player = Players:FindFirstChild(playerName)
                    if player and player ~= LocalPlayer and not IsWhitelisted(player) then
                        local exists = false
                        for _, p in ipairs(allPlayers) do
                            if p.Name == playerName then
                                exists = true
                                break
                            end
                        end
                        if not exists then
                            table.insert(allPlayers, player)
                        end
                    end
                end
            end
        end
    end
    return allPlayers
end

local function GetPlayerHRP(player)
    if not player then return nil end
    local char = player.Character
    if char and char.Parent == Workspace then
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if hrp then return hrp end
    end
    local plotItems = Workspace:FindFirstChild("PlotItems")
    if plotItems then
        for _, plot in ipairs(plotItems:GetChildren()) do
            if plot:IsA("Folder") then
                local playersInPlot = plot:FindFirstChild("PlayersInPlots")
                if playersInPlot and playersInPlot:FindFirstChild(player.Name) then
                    for _, child in ipairs(plot:GetDescendants()) do
                        if child:IsA("Model") and child.Name == player.Name then
                            local hrp = child:FindFirstChild("HumanoidRootPart")
                            if hrp then return hrp end
                        end
                    end
                end
            end
        end
    end
    for _, model in ipairs(Workspace:GetDescendants()) do
        if model:IsA("Model") and model.Name == player.Name then
            local hrp = model:FindFirstChild("HumanoidRootPart")
            if hrp then return hrp end
        end
    end
    return nil
end

local function CreateBodies(targetRoot, pos)
    if not targetRoot or not pos then return end
    if currentBodyPosition then
        currentBodyPosition:Destroy()
        currentBodyPosition = nil
    end
    if currentBodyGyro then
        currentBodyGyro:Destroy()
        currentBodyGyro = nil
    end
    currentBodyPosition = Instance.new("BodyPosition")
    currentBodyPosition.MaxForce = Vector3.new(1e8, 1e8, 1e8)
    currentBodyPosition.P = 100000
    currentBodyPosition.D = 1000
    currentBodyPosition.Position = pos
    currentBodyPosition.Parent = targetRoot
    currentBodyGyro = Instance.new("BodyGyro")
    currentBodyGyro.MaxTorque = Vector3.new(1e8, 1e8, 1e8)
    currentBodyGyro.P = 100000
    currentBodyGyro.D = 1000
    currentBodyGyro.CFrame = CFrame.new(pos)
    currentBodyGyro.Parent = targetRoot
end

local function CleanupBodies()
    if currentBodyPosition then
        currentBodyPosition:Destroy()
        currentBodyPosition = nil
    end
    if currentBodyGyro then
        currentBodyGyro:Destroy()
        currentBodyGyro = nil
    end
end

local BlobmanBeta = {}

function BlobmanBeta.getLocalChar()
    return LocalPlayer.Character
end

function BlobmanBeta.getLocalRoot()
    local char = BlobmanBeta.getLocalChar()
    return char and (char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso"))
end

function BlobmanBeta.getLocalHum()
    local char = BlobmanBeta.getLocalChar()
    return char and char:FindFirstChildOfClass("Humanoid")
end

function BlobmanBeta.getInv()
    return Workspace:FindFirstChild(LocalPlayer.Name .. "SpawnedInToys")
end

function BlobmanBeta.SetNetworkOwner(part)
    ReplicatedStorage.GrabEvents.SetNetworkOwner:FireServer(part, BlobmanBeta.getLocalRoot().CFrame)
end

function BlobmanBeta.ungrab(part)
    ReplicatedStorage.GrabEvents.DestroyGrabLine:FireServer(part)
end

function BlobmanBeta.getBlobman()
    local inv = BlobmanBeta.getInv()
    if inv then
        local v = inv:FindFirstChild("CreatureBlobman")
        if v and v:FindFirstChild("VehicleSeat") then
            return v
        end
    end
    local plotItems = Workspace:FindFirstChild("PlotItems")
    if plotItems then
        for _, p in plotItems:GetChildren() do
            local m = p:FindFirstChild("CreatureBlobman")
            if m and m:FindFirstChild("PlayerValue") and m.PlayerValue.Value == LocalPlayer.Name then
                return m
            end
        end
    end
    return nil
end

function BlobmanBeta.spawnBlobman()
    local success, result = pcall(function()
        ReplicatedStorage.MenuToys.SpawnToyRemoteFunction:InvokeServer("CreatureBlobman", BlobmanBeta.getLocalRoot().CFrame, Vector3.zero)
    end)
    if not success then
        warn("Failed to spawn Blobman: " .. tostring(result))
        return nil
    end
    task.wait(1.0)
    return BlobmanBeta.getBlobman()
end

function BlobmanBeta.destroyBlobman()
    local blob = BlobmanBeta.getBlobman()
    if blob then
        ReplicatedStorage.MenuToys.DestroyToy:FireServer(blob)
    end
end

function BlobmanBeta.isSittingOnBlobman()
    local hum = BlobmanBeta.getLocalHum()
    if not hum then
        return false
    end
    local blob = BlobmanBeta.getBlobman()
    if not blob or not blob:FindFirstChild("VehicleSeat") then
        return false
    end
    return hum.Sit and hum.SeatPart == blob.VehicleSeat
end

function BlobmanBeta.ensureSitBlobman()
    local blob = BlobmanBeta.getBlobman()
    if not blob or not blob:FindFirstChild("VehicleSeat") then
        return false
    end
    local seat = blob.VehicleSeat
    local hum = BlobmanBeta.getLocalHum()
    if hum and not hum.Sit then
        seat:Sit(hum)
        task.wait(0.6)
    end
    return BlobmanBeta.isSittingOnBlobman()
end

function BlobmanBeta.blobGrab(blob, target, side)
    if not blob then
        return
    end
    local detector = blob:FindFirstChild(side .. "Detector")
    if not detector then
        return
    end
    local weld = detector:FindFirstChild(side .. "Weld")
    if not weld then
        return
    end
    local script = blob:FindFirstChild("BlobmanSeatAndOwnerScript", true)
    if not script then
        return
    end
    local remote = script:FindFirstChild("CreatureGrab")
    if remote then
        pcall(function()
            remote:FireServer(detector, target, weld)
        end)
    end
end

function BlobmanBeta.blobDrop(blob, target, side)
    if not blob then
        return
    end
    local detector = blob:FindFirstChild(side .. "Detector")
    if not detector then
        return
    end
    local script = blob:FindFirstChild("BlobmanSeatAndOwnerScript", true)
    if not script then
        return
    end
    local remote = script:FindFirstChild("CreatureDrop")
    if remote then
        pcall(function()
            remote:FireServer(detector, target)
        end)
    end
end

function BlobmanBeta.blobKick(blob, targetRoot, side)
    BlobmanBeta.blobGrab(blob, BlobmanBeta.getLocalRoot(), side)
    task.wait(0.08)
    BlobmanBeta.SetNetworkOwner(targetRoot)
    task.wait(0.08)
    targetRoot.CFrame = targetRoot.CFrame + Vector3.new(0, 16, 0)
    task.wait(0.08)
    BlobmanBeta.ungrab(targetRoot)
    task.wait(0.08)
    BlobmanBeta.blobGrab(blob, targetRoot, side)
    task.wait(0.08)
    BlobmanBeta.blobDrop(blob, targetRoot, side)
    task.wait(0.08)
    BlobmanBeta.ungrab(targetRoot)
end

local function KickWithBlobman(targetPlayer)
    local success, result = pcall(function()
        local targetChar = targetPlayer.Character
        if not targetChar then
            return false, "Target has no character"
        end
        local targetRoot = targetChar:FindFirstChild("HumanoidRootPart")
        if not targetRoot then
            return false, "Target has no HumanoidRootPart"
        end
        local blob = BlobmanBeta.getBlobman()
        if not blob then
            blob = BlobmanBeta.spawnBlobman()
            if not blob then
                return false, "Failed to spawn Blobman"
            end
        end
        if not BlobmanBeta.isSittingOnBlobman() then
            BlobmanBeta.ensureSitBlobman()
            if not BlobmanBeta.isSittingOnBlobman() then
                return false, "Not sitting on Blobman"
            end
        end
        local myRoot = BlobmanBeta.getLocalRoot()
        if myRoot and targetRoot then
            local oldPos = myRoot.CFrame
            myRoot.CFrame = targetRoot.CFrame
            task.wait(0.07)
            BlobmanBeta.blobKick(blob, targetRoot, "Left")
            task.wait(0.25)
            myRoot.CFrame = oldPos
        end
        BlobmanBeta.destroyBlobman()
        return true, "Kicked " .. targetPlayer.Name .. "!"
    end)
    if not success then
        warn("KickWithBlobman error: " .. tostring(result))
        return false, "Error occurred"
    end
    return true, result
end

-- Kick All 機能
local currentBlob = nil
local isActive = false
local playerStatus = {}

local function ResetTargets()
    for id, status in pairs(playerStatus) do
        if status == "Targeting" then playerStatus[id] = nil end
    end
end

local function KickAll()
    if isActive then return end
    isActive = true
    
    local allPlayers = GetAllPlayers()
    if #allPlayers == 0 then 
        isActive = false
        return 
    end

    for _, targetPlayer in ipairs(allPlayers) do
        playerStatus[targetPlayer.UserId] = "Targeting"
    end
    
    local rootPart = BlobmanBeta.getLocalRoot()
    if rootPart then
        local spawnPos = rootPart.CFrame * CFrame.new(0, 0, -5)
        ReplicatedStorage.MenuToys.SpawnToyRemoteFunction:InvokeServer("CreatureBlobman", spawnPos, Vector3.new(0, 127, 0))
    end
    task.wait(0.5)
    
    local toyFolder = Workspace:FindFirstChild(LocalPlayer.Name .. "SpawnedInToys")
    currentBlob = toyFolder and toyFolder:FindFirstChild("CreatureBlobman")
    if not currentBlob then 
        ResetTargets()
        isActive = false
        return 
    end
    
    local vehicleSeat = currentBlob:FindFirstChild("VehicleSeat")
    if vehicleSeat and LocalPlayer.Character then
        local humanoid = LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
        if humanoid then
            vehicleSeat:Sit(humanoid)
        end
    end
    task.wait(0.3)
    
    local myRoot = BlobmanBeta.getLocalRoot()
    if not myRoot then 
        ResetTargets()
        isActive = false
        return 
    end
    
    local tpWait = 0.025
    
    for _, targetPlayer in ipairs(allPlayers) do
        local targetRoot = targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart")
        if targetRoot then
            myRoot.CFrame = targetRoot.CFrame
            myRoot.AssemblyAngularVelocity = Vector3.zero
            task.wait(tpWait)
            
            local distance = (myRoot.Position - targetRoot.Position).Magnitude
            if distance <= 35 then
                for i = 1, 8 do
                    if not targetRoot.Parent or not currentBlob.Parent then break end
                    pcall(function()
                        currentBlob.BlobmanSeatAndOwnerScript.CreatureGrab:FireServer(
                            currentBlob.LeftDetector, targetRoot, currentBlob.LeftDetector.LeftWeld
                        )
                        currentBlob.BlobmanSeatAndOwnerScript.CreatureRelease:FireServer(currentBlob.LeftDetector.LeftWeld)
                    end)
                    task.wait()
                end
            end
        end
    end
    
    myRoot.CFrame = CFrame.new(0, 100, 0)
    myRoot.AssemblyLinearVelocity = Vector3.zero
    task.wait(0.1)
    
    for _, part in ipairs(currentBlob:GetDescendants()) do
        if part:IsA("BasePart") then pcall(function() part.Anchored = true end) end
    end
    task.wait(0.1)
    
    local radius = 15
    for i, targetPlayer in ipairs(allPlayers) do
        local targetRoot = targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart")
        if targetRoot then
            local angle = math.rad((i - 1) * (360 / #allPlayers))
            local x = radius * math.cos(angle)
            local z = radius * math.sin(angle)
            targetRoot.CFrame = CFrame.new(x, 110, z)
        end
    end
    task.wait(0.1)
    
    for _ = 1, 2 do
        for _, targetPlayer in ipairs(allPlayers) do
            local targetRoot = targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart")
            if targetRoot then
                task.spawn(function()
                    pcall(function()
                        ReplicatedStorage.GrabEvents.SetNetworkOwner:FireServer(targetRoot, CFrame.new(targetRoot.Position))
                        ReplicatedStorage.GrabEvents.DestroyGrabLine:FireServer(targetRoot)
                    end)
                end)
            end
        end
        task.wait(0.1)
    end
    
    task.wait(0.3)
    
    for _, targetPlayer in ipairs(allPlayers) do
        local targetRoot = targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart")
        if targetRoot then
            task.spawn(function()
                pcall(function()
                    currentBlob.BlobmanSeatAndOwnerScript.CreatureGrab:FireServer(
                        currentBlob.LeftDetector, targetRoot, currentBlob.LeftDetector.LeftWeld
                    )
                    currentBlob.BlobmanSeatAndOwnerScript.CreatureGrab:FireServer(
                        currentBlob.RightDetector, targetRoot, currentBlob.RightDetector.RightWeld
                    )
                end)
            end)
        end
    end
    
    task.wait(0.1)

    for _, targetPlayer in ipairs(allPlayers) do
        if targetPlayer and targetPlayer.Parent == Players then
            playerStatus[targetPlayer.UserId] = "Kicked"
        end
    end
    
    for _, part in ipairs(currentBlob:GetDescendants()) do
        if part:IsA("BasePart") then pcall(function() part.Anchored = false end) end
    end
    
    task.wait(1)
    isActive = false
end

local function teleportToPlayer(myHrp, targetHrp)

        myHrp.CFrame = targetHrp.CFrame * CFrame.new(0, 5, 5)

local function destroyLineOnPlayer(hrp)
    if not setNetworkOwnerEvent or not destroyGrabLineEvent then
        return
    end
    pcall(function()
        setNetworkOwnerEvent:FireServer(hrp, CFrame.new(0, 1e9, 0))
        task.wait()
        destroyGrabLineEvent:FireServer(hrp)
    end)
end

local function startLagSpam()
    if lagConnection then
        lagConnection:Disconnect()
        lagConnection = nil
    end
    local targetRate = 85
    local frameRate = 60
    local basePerFrame = math.floor(targetRate / frameRate)
    local remainder = targetRate - (basePerFrame * frameRate)
    local frameCounter = 0
    lagConnection = RunService.Heartbeat:Connect(function()
        frameCounter = frameCounter + 1
        local sendCount = basePerFrame
        if frameCounter <= remainder then
            sendCount = sendCount + 1
        end
        local spawnLocation = Workspace:FindFirstChild("SpawnLocation")
            or Workspace:FindFirstChild("Spawn")
            or (LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart"))
        if spawnLocation then
            for i = 1, sendCount do
                pcall(function()
                    local randomX = math.random(-2010000000, 2000000001)
                    local randomZ = math.random(-2008100000, 2000200000)
                    setNetworkOwnerEvent:FireServer(spawnLocation, CFrame.new(randomX, 0, randomZ))
                end)
            end
        end
    end)
end

local function stopLagSpam()
    if lagConnection then
        lagConnection:Disconnect()
        lagConnection = nil
    end
end

local function teleportToCenter(hrp, centerPosition, height)
    if not hrp then
        return
    end
    pcall(function()
        hrp.CFrame = CFrame.new(centerPosition.X, height, centerPosition.Z)
        hrp.AssemblyLinearVelocity = Vector3.zero
    end)
end

local function executeTargetLag()
    if isTargetLagRunning then
        return
    end
    isTargetLagRunning = true
    targetLagTask = task.spawn(function()
        startLagSpam()
        local height = 35
        task.wait(1)
        local players = GetAllPlayersIncludingInPlot()
        if #players == 0 then
            task.wait(10)
            stopLagSpam()
            isTargetLagRunning = false
            return
        end
        local myChar = LocalPlayer.Character
        local myHrp = myChar and myChar:FindFirstChild("HumanoidRootPart")
        if not myHrp then
            task.wait(10)
            stopLagSpam()
            isTargetLagRunning = false
            return
        end
        local playerData = {}
        for _, plr in ipairs(players) do
            local hrp = GetPlayerHRP(plr)
            if hrp then
                table.insert(playerData, {player = plr, hrp = hrp})
            end
        end
        local spawnLocation = Workspace:FindFirstChild("SpawnLocation")
            or Workspace:FindFirstChild("Spawn")
            or Workspace:FindFirstChild("SpawnLocation1")
        local centerX = spawnLocation and spawnLocation.Position.X or 0
        local centerZ = spawnLocation and spawnLocation.Position.Z or 0
        local centerPosition = Vector3.new(centerX, height, centerZ)
        teleportToCenter(myHrp, centerPosition, height)
        for _, data in ipairs(playerData) do
            teleportToPlayer(myHrp, data.hrp)
            task.wait(0.2)
            spamOwnership(data.hrp)
            task.wait()
        end
        local radius = 80
        local angleStep = (math.pi * 2) / #playerData
        for idx, data in ipairs(playerData) do
            local angle = (idx - 1) * angleStep
            local x = math.cos(angle) * radius
            local z = math.sin(angle) * radius
            pcall(function()
                data.hrp.CFrame = CFrame.new(centerX + x, height, centerZ + z)
                data.hrp.AssemblyLinearVelocity = Vector3.zero
            end)
            local bp = Instance.new("BodyPosition")
            bp.MaxForce = Vector3.new(9e9, 9e9, 9e9)
            bp.P = 50000000000
            bp.Position = Vector3.new(centerX + x, height, centerZ + z)
            bp.Parent = data.hrp
            task.delay(2, function()
                pcall(function()
                    bp:Destroy()
                end)
            end)
            task.wait()
        end
        startLagSpam()
        task.wait(0.1)
        teleportToCenter(myHrp, centerPosition, height)
        for i = 1, 8 do
            for _, data in ipairs(playerData) do

local function CounterAttack(attacker)
    if not attacker or not attacker.Character then
        return
    end
    local tChar = attacker.Character
    local tHrp = tChar:FindFirstChild("HumanoidRootPart")
    local tHum = tChar:FindFirstChildOfClass("Humanoid")
    if not tHrp or not tHum or tHum.Health <= 0 then
        return
    end
    if CounterMode == "Death" then
        tHum.Health = 0
        tHum.BreakJointsOnDeath = false
        tHum:ChangeState(Enum.HumanoidStateType.Dead)
    elseif CounterMode == "Kick" then
        if setNetworkOwnerEvent then
            setNetworkOwnerEvent:FireServer(tHrp, tHrp.CFrame)
        end
        local bv = Instance.new("BodyVelocity")
        bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
        bv.Velocity = Vector3.new(0, 1000, 0)
        bv.Parent = tHrp
        Debris:AddItem(bv, 0.5)
        task.wait(0.1)
        if destroyGrabLineEvent then
            destroyGrabLineEvent:FireServer(tHrp)
        end
    elseif CounterMode == "Freeze" then
        tHum.WalkSpeed = 0
        tHum.JumpPower = 0
        tHum.PlatformStand = true
    elseif CounterMode == "Repulsion" then
        if setNetworkOwnerEvent then
            setNetworkOwnerEvent:FireServer(tHrp, tHrp.CFrame)
        end
        local bv = Instance.new("BodyVelocity")
        bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
        local dir = (tHrp.Position - LocalPlayer.Character.HumanoidRootPart.Position).Unit
        bv.Velocity = Vector3.new(dir.X * 200, 100, dir.Z * 200)
        bv.Parent = tHrp
        Debris:AddItem(bv, 0.3)
        if destroyGrabLineEvent then
            destroyGrabLineEvent:FireServer(tHrp)
        end
    end
end

-- ToyModオブジェクト検索
local function findToyModObjects()
    local toys = {}
    for _, item in ipairs(Workspace:GetDescendants()) do
        if item:IsA("Model") and item.Name == ToyMod.ObjectID then
            local alreadyAdded = false
            for _, existingToy in ipairs(toys) do
                if existingToy == item then
                    alreadyAdded = true
                    break
                end
            end
            if not alreadyAdded then
                table.insert(toys, item)
            end
        end
    end
    table.sort(toys, function(a, b)
        return a.Name < b.Name
    end)
    return toys
end

local function getToyModPrimaryPart(model)
    if model.PrimaryPart then

    end
    local potentialParts = {
        "Handle",

        "Sparkler",

        "Snowflake",

        "Tractor"
    }
    for _, partName in ipairs(potentialParts) do
        local part = model:FindFirstChild(partName)
        if part and part:IsA("BasePart") then

    for _, child in ipairs(model:GetChildren()) do

-- ToyMod物理アタッチ
local function attachToyModPhysics(part, pValue, dValue)
    if not part then
        return nil, nil
    end
    local existingBG = part:FindFirstChildOfClass("BodyGyro")
    local existingBP = part:FindFirstChildOfClass("BodyPosition")
    if existingBG and existingBP then
        return existingBG, existingBP
    end
    if existingBG then
        existingBG:Destroy()
    end
    if existingBP then
        existingBP:Destroy()
    end
    local BP = Instance.new("BodyPosition")
    local BG = Instance.new("BodyGyro")
    BP.P = pValue or 15000
    BP.D = dValue or 200
    BP.MaxForce = Vector3.new(1, 1, 1) * 1e10
    BP.Parent = part
    BG.P = pValue or 15000
    BG.D = dValue or 200
    BG.MaxTorque = Vector3.new(1, 1, 1) * 1e10
    BG.Parent = part
    return BG, BP
end

-- Featherポイント作成
local function createFeatherPoints(count)
    local points = {}
    if count == 0 then
        return points
    end
    local halfCount = math.floor(count / 2)
    local isOdd = count % 2 == 1
    for i = 1, count do
        local x = isOdd and (i - math.ceil(count / 2)) * ToyMod.Feather.Spacing or (i - halfCount - 0.5) * ToyMod.Feather.Spacing
        local part = Instance.new("Part")
        part.CanCollide = false
        part.Anchored = true
        part.Transparency = 1
        part.Size = Vector3.new(4, 1, 4)
        part.Parent = Workspace
        points[i] = {
            offsetX = x,
            part = part,
            assignedToy = nil,
            distanceRank = 0
        }
    end
    return points
end

-- Heartポイント作成
local function createHeartPoints(count)
    local points = {}
    if count == 0 then
        return points
    end
    for i = 1, count do
        local t = (i - 1) * (2 * math.pi / count)
        local part = Instance.new("Part")

        part.Anchored = true
        part.Transparency = 1
        part.Size = Vector3.new(4, 1, 4)
        part.Parent = Workspace
        points[i] = {
            angle = t,
            part = part,
            assignedToy = nil
        }
    end
    return points
end

-- Starポイント作成
local function createStarPoints(count)
    local points = {}
    if count == 0 then
        return points
    end
    for i = 1, count do
        local starIndex = (i - 1) % 10
        local isOuter = starIndex % 2 == 0
        local part = Instance.new("Part")
        part.CanCollide = false
        part.Anchored = true
        part.Transparency = 1
        part.Size = Vector3.new(4, 1, 4)
        part.Parent = Workspace
        points[i] = {
            starIndex = starIndex,
            isOuter = isOuter,
            part = part,
            assignedToy = nil
        }
    end
    return points
end

-- Heart位置計算
local function getHeartPosition(t, size, pulse, scale, verticalStretch)
    local baseScale = size / 20
    local currentScale = baseScale * (scale or 1)
    local x = 16 * (math.sin(t) ^ 3) * currentScale
    local y = (13 * math.cos(t) - 5 * math.cos(2 * t) - 2 * math.cos(3 * t) - math.cos(4 * t)) * currentScale
    if verticalStretch and verticalStretch > 1 then
        y = y * verticalStretch
    end
    if pulse > 0 then
        local pulseFactor = 1 + (pulse * 0.1)
        x = x * pulseFactor
        y = y * pulseFactor
    end
    return x, y
end

-- Star位置計算
local function getStarPosition(starIndex, isOuter, outerRadius, innerRadius, time, rotationSpeed, twinkleSpeed)
    local anglePerPoint = 2 * math.pi / 5
    local pointAngle = starIndex * (anglePerPoint / 2)
    local radius = isOuter and outerRadius or innerRadius
    local rotationAngle = pointAngle + (time * rotationSpeed)
    local twinkle = math.sin(time * twinkleSpeed + starIndex) * 0.2
    local finalRadius = radius * (1 + twinkle)
    local x = math.cos(rotationAngle) * finalRadius
    local z = math.sin(rotationAngle) * finalRadius
    return x, z, pointAngle
end

-- ToyModをポイントに割り当て
local function assignToyModToysToPoints(points)
    ToyModAssignedToys = {}
    for i = 1, math.min(#ToyModToys, #points) do
        local toy = ToyModToys[i]
        if toy and toy:IsA("Model") and toy.Name == ToyMod.ObjectID then
            local primaryPart = getToyModPrimaryPart(toy)
            if primaryPart then
                for _, child in ipairs(toy:GetChildren()) do
                    if child:IsA("BasePart") then
                        child.CanCollide = false
                        child.CanTouch = false
                        child.Anchored = false
                    end
                end
                local BG, BP = attachToyModPhysics(primaryPart)
                local toyTable = {
                    BG = BG,
                    BP = BP,
                    Pallet = primaryPart,
                    Model = toy,
                    PointIndex = i
                }
                if points[i].offsetX then
                    toyTable.offsetX = points[i].offsetX
                    toyTable.distanceRank = points[i].distanceRank
                end
                if points[i].angle then
                    toyTable.baseAngle = points[i].angle
                end
                if points[i].starIndex then
                    toyTable.starIndex = points[i].starIndex
                    toyTable.isOuter = points[i].isOuter
                end
                points[i].assignedToy = toyTable
                table.insert(ToyModAssignedToys, toyTable)
            end
        end
    end
    return ToyModAssignedToys
end

-- ToyModループ開始
local function startToyModLoop()
    if ToyModLoopConn then
        ToyModLoopConn:Disconnect()
        ToyModLoopConn = nil
    end
    ToyModTime = 0
    ToyModLoopConn = RunService.RenderStepped:Connect(function(dt)
        if not ToyMod.Enabled then
            return
        end
        local char = LocalPlayer.Character
        if not char then
            return
        end
        local humanoidRootPart = char:FindFirstChild("HumanoidRootPart")
        local torso = char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso")
        if not humanoidRootPart or not torso then
            return
        end
        ToyModTime = ToyModTime + dt
        local basePosition = ToyMod.FollowPlayer and torso.Position or torso.Position
        local mode = ToyMod.CurrentMode
        if mode == "Feather" then
            local rightVector = humanoidRootPart.CFrame.RightVector
            local backVector = -humanoidRootPart.CFrame.LookVector
            local basePos = torso.Position + Vector3.new(0, ToyMod.Feather.HeightOffset, 0) + (backVector * ToyMod.Feather.BackwardOffset)
            for i, point in ipairs(ToyModPoints) do
                if point.assignedToy and point.assignedToy.BP and point.assignedToy.BG then
                    local toy = point.assignedToy
                    local targetPosition = basePos + (rightVector * toy.offsetX)
                    local amplitude = ToyMod.Feather.BaseAmplitude * toy.distanceRank
                    local waveMovement = math.sin(ToyModTime * ToyMod.Feather.WaveSpeed) * amplitude
                    local finalPosition = targetPosition + Vector3.new(0, waveMovement, 0)
                    if point.part then
                        point.part.Position = finalPosition
                    end
                    toy.BP.Position = finalPosition
                    local backYRotation = math.atan2(-humanoidRootPart.CFrame.LookVector.X, -humanoidRootPart.CFrame.LookVector.Z)
                    local baseCFrame = CFrame.new(finalPosition) * CFrame.Angles(0, backYRotation, 0)
                    local tiltedCFrame = baseCFrame * CFrame.Angles(math.rad(-ToyMod.Feather.TiltAngle), 0, 0)
                    toy.BG.CFrame = toy.BG.CFrame:Lerp(tiltedCFrame, 0.3)
                end
            end
        elseif mode == "Heart" then
            local pulseEffect = 0
            if ToyMod.Heart.PulseSpeed > 0 then
                pulseEffect = math.sin(ToyModTime * ToyMod.Heart.PulseSpeed) * ToyMod.Heart.PulseAmplitude
            end
            for i, point in ipairs(ToyModPoints) do
                if point.assignedToy and point.assignedToy.BP and point.assignedToy.BG then
                    local toy = point.assignedToy
                    local currentAngle = toy.baseAngle + (ToyModTime * ToyMod.Heart.RotationSpeed)
                    local x, y = getHeartPosition(currentAngle, ToyMod.Heart.Size, pulseEffect, 1, 1)
                    local heightOffset = ToyMod.Heart.Height + (math.sin(currentAngle * 2) * 0.5)
                    local localPos = Vector3.new(x, heightOffset, y)
                    local targetPosition = basePosition + localPos
                    if point.part then
                        point.part.Position = targetPosition
                    end
                    toy.BP.Position = targetPosition
                    toy.BG.CFrame = CFrame.new(targetPosition) * CFrame.Angles(-math.rad(90), 0, 0)
                end
            end
        elseif mode == "BigHeart" then
            local pulseEffect = 0
            if ToyMod.BigHeart.PulseSpeed > 0 then
                pulseEffect = math.sin(ToyModTime * ToyMod.BigHeart.PulseSpeed) * ToyMod.BigHeart.PulseAmplitude
            end
            for i, point in ipairs(ToyModPoints) do
                if point.assignedToy and point.assignedToy.BP and point.assignedToy.BG then
                    local toy = point.assignedToy
                    local currentAngle = toy.baseAngle + (ToyModTime * ToyMod.BigHeart.RotationSpeed)
                    local x, y = getHeartPosition(currentAngle, ToyMod.BigHeart.Size, pulseEffect, ToyMod.BigHeart.HeartScale, ToyMod.BigHeart.VerticalStretch)
                    local heightOffset = ToyMod.BigHeart.Height + (math.sin(currentAngle * 2) * 1.0)
                    local localPos = Vector3.new(x, heightOffset, y)
                    local targetPosition = basePosition + localPos
                    if point.part then
                        point.part.Position = targetPosition
                    end
                    toy.BP.Position = targetPosition
                    toy.BG.CFrame = CFrame.new(targetPosition) * CFrame.Angles(-math.rad(90), 0, 0)
                end
            end
        elseif mode == "Star" then
            for i, point in ipairs(ToyModPoints) do
                if point.assignedToy and point.assignedToy.BP and point.assignedToy.BG then
                    local toy = point.assignedToy
                    local x, z, pointAngle = getStarPosition(toy.starIndex, toy.isOuter, ToyMod.Star.OuterRadius, ToyMod.Star.InnerRadius, ToyModTime, ToyMod.Star.RotationSpeed, ToyMod.Star.TwinkleSpeed)
                    local heightVariation = math.sin(pointAngle * 3) * 0.5
                    local finalHeight = ToyMod.Star.Height + heightVariation
                    local localPos = Vector3.new(x, finalHeight, z)
                    local targetPosition = basePosition + localPos
                    if point.part then
                        point.part.Position = targetPosition
                    end
                    toy.BP.Position = targetPosition
                    local direction = (targetPosition - basePosition).Unit
                    if direction.Magnitude > 0 then
                        toy.BG.CFrame = CFrame.lookAt(targetPosition, targetPosition + direction)
                    end
                end
            end
        end
    end)
end

-- ToyModループ停止
local function stopToyModLoop()
    if ToyModLoopConn then
        ToyModLoopConn:Disconnect()
        ToyModLoopConn = nil
    end
    for _, point in ipairs(ToyModPoints) do
        if point.part then
            point.part:Destroy()
        end
        if point.assignedToy then
            if point.assignedToy.BG then
                point.assignedToy.BG:Destroy()
            end
            if point.assignedToy.BP then
                point.assignedToy.BP:Destroy()
            end
        end
    end
    ToyModPoints = {}
    ToyModAssignedToys = {}
end

-- ToyMod切り替え
local function toggleToyMod(state)
    if state then
        if ToyMod.CurrentMode == "None" then
            ToyMod.Enabled = false
            return
        end
        ToyMod.Enabled = true
        ToyModToys = findToyModObjects()
        local mode = ToyMod.CurrentMode
        local count = 0
        if mode == "Feather" then
            ToyModPoints = createFeatherPoints(math.min(#ToyModToys, ToyMod.Feather.MaxObjects))
            count = #ToyModPoints
        elseif mode == "Heart" then
            ToyModPoints = createHeartPoints(math.min(#ToyModToys, ToyMod.Heart.ObjectCount))
            count = #ToyModPoints
        elseif mode == "BigHeart" then
            ToyModPoints = createBigHeartPoints(math.min(#ToyModToys, ToyMod.BigHeart.ObjectCount))
            count = #ToyModPoints
        elseif mode == "Star" then
            ToyModPoints = createStarPoints(math.min(#ToyModToys, ToyMod.Star.ObjectCount))
            count = #ToyModPoints
        end
        ToyModAssignedToys = assignToyModToysToPoints(ToyModPoints)
        startToyModLoop()
    else
        ToyMod.Enabled = false
        stopToyModLoop()
    end
end
