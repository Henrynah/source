        TweenInfo.new(0.5, Enum.EasingStyle.Quad, Enum.EasingDirection.InOut),
        {
            Size = UDim2.new(0, 300, 0, 300),
            Position = UDim2.new(0.5, -150, 0.5, -150)
        }
    )
    
    bgTween:Play()
    wait(0.2)
    imageTween1:Play()
    imageTween1.Completed:Wait()
    imageTween2:Play()
    imageTween2.Completed:Wait()
    
    wait(1)

        frame,
        TweenInfo.new(0.8, Enum.EasingStyle.Quad, Enum.EasingDirection.In),

            BackgroundTransparency = 1

        imageLabel,
        TweenInfo.new(0.8, Enum.EasingStyle.Quad, Enum.EasingDirection.In),
        {
            ImageTransparency = 1
        }
    )
    
    fadeOutTween:Play()
    imageFadeOut:Play()
    fadeOutTween.Completed:Wait()
    
    screenGui:Destroy()
end

local function Internal_HsvToRgb(h, s, v)
    local c, x = v * s, v * s * (1 - math.abs(math.fmod(h / 60, 2) - 1))
    local m = v - c
    local r, g, b = 0, 0, 0
    if h < 60 then r,g,b=c,x,0 elseif h < 120 then r,g,b=x,c,0 elseif h < 180 then r,g,b=0,c,x
    elseif h < 240 then r,g,b=0,x,c elseif h < 300 then r,g,b=x,0,c else r,g,b=c,0,x end
    return Color3.new(r+m, g+m, b+m)
end

local function Velocity(part, vel)
    if not part or not part.Parent then return end
    local bv = Instance.new("BodyVelocity")
    bv.MaxForce = Vector3.new(1e8, 1e8, 1e8)
    bv.Velocity = vel

local function MoveTo(part, targetCFrame)
    if not part or not part.Parent then return end
    for _, v in ipairs(part.Parent:GetDescendants()) do
        if v:IsA("BasePart") then v.CanCollide = false end
    end
    local b = Instance.new("BodyPosition")
    b.MaxForce = Vector3.new(1e8, 1e8, 1e8)
    b.Position = targetCFrame.Position
    b.P, b.D = 2e4, 5e3
    b.Parent = part
    task.spawn(function()
        task.wait(1)
        pcall(function()
            b:Destroy()
            for _, v in ipairs(part.Parent:GetDescendants()) do
                if v:IsA("BasePart") then v.CanCollide = true end
            end
        end)
    end)
end

local function Loop2_lookAt(from, to)
    local direction = (to - from).Unit
    local right = direction:Cross(Vector3.new(0, 1, 0))
    local up = right:Cross(direction)
    return CFrame.fromMatrix(from, right, up)
end

local function Loop2_EnableNoclip()
    if not Config.Loop2.Loop2_NoclipConnection then
        Config.Loop2.Loop2_Noclipping = false
        Config.Loop2.Loop2_NoclipConnection = RunService.Stepped:Connect(function()
            if Config.Loop2.Loop2_Noclipping == false and LP.Character then
                for _, part in pairs(LP.Character:GetChildren()) do
                    if part:IsA("BasePart") and part.CanCollide and part.Name ~= "floatName" then
                        part.CanCollide = false
                    end
                end
            end
            task.wait(0.21)
        end)
    end
end

local function Loop2_DisableNoclip()
    if Config.Loop2.Loop2_NoclipConnection then
        Config.Loop2.Loop2_NoclipConnection:Disconnect()
        Config.Loop2.Loop2_NoclipConnection = nil
    end
    Config.Loop2.Loop2_Noclipping = true
end

local function Loop2_SNOWship(part)
    if part and typeof(part) == "Instance" then
        local distance = LP:DistanceFromCharacter(part.Position)
        if LP.Character and LP.Character:FindFirstChild("HumanoidRootPart") and distance <= 30 then
            SetNetworkOwner(part, Loop2_lookAt(LP.Character.HumanoidRootPart.Position, part.Position))
        end
    end
end

local function Loop2_TeleportPlayer(cf)
    local character = LP.Character
    if character and typeof(cf) == "CFrame" then
        local root = character:FindFirstChild("HumanoidRootPart")
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        if root and humanoid then
            root.CFrame = root.CFrame.Rotation + cf.Position
            if humanoid.SeatPart == nil or tostring(humanoid.SeatPart.Parent) ~= "CreatureBlobman" then
                humanoid.Sit = false
            end
        end
    end
end

local function Loop2_CreateSkyVelocity(part)
    if not part:FindFirstChild("SkyVelocity") then
        local bv = Instance.new("BodyVelocity", part)
        bv.Name = "SkyVelocity"
        bv.Velocity = Vector3.new(0, 100000000000000, 0)
        bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    end
end

local function Loop2_CheckNetworkOwnership(player)
    if typeof(player) == "Instance" and player:IsA("Player") and player.Character then
        local head = player.Character:FindFirstChild("Head")
        if head and head:FindFirstChild("PartOwner") and head.PartOwner.Value == LP.Name then
            return true
        end
    end
    return false
end

local function Loop2_GetPlayerCFrame()
    if LP.Character and LP.Character:FindFirstChild("HumanoidRootPart") then
        return LP.Character.HumanoidRootPart.CFrame
    end
end

local function Loop2_IsValidTarget(player)
    if typeof(player) == "Instance" and player ~= LP then
        if player:IsA("Player") and player.Character then
            if player.Character:IsDescendantOf(Workspace) then
                local root = player.Character:FindFirstChild("HumanoidRootPart")
                local humanoid = player.Character:FindFirstChildOfClass("Humanoid")
                if root and humanoid and humanoid.Health > 0 then
                    if player:FindFirstChild("InPlot") and player.InPlot.Value then return false end
                    if humanoid.FloorMaterial == Enum.Material.Air then return false end
                    return true
                end
            end
        end
    end
    return false
end

local function Loop2_IsValidKickTarget(player)
    if typeof(player) == "Instance" and player ~= LP then
        if player:IsA("Player") and player.Character then
            if player.Character:IsDescendantOf(Workspace) then
                local root = player.Character:FindFirstChild("HumanoidRootPart")
                local humanoid = player.Character:FindFirstChildOfClass("Humanoid")
                if root and humanoid and humanoid.Health > 0 then
                    if player:FindFirstChild("InPlot") and player.InPlot.Value then return false end
                    if humanoid.FloorMaterial == Enum.Material.Air then return false end
                    return true
                end
            end
        end
    end
    return false
end

local function Loop2_KillPlayer(targetPlayer)
    if not Loop2_IsValidTarget(targetPlayer) then return end

    local root = targetPlayer.Character:FindFirstChild("HumanoidRootPart")
    local humanoid = targetPlayer.Character:FindFirstChildOfClass("Humanoid")
    if not root or not humanoid then return end

    local initialPos = LP.Character.HumanoidRootPart.CFrame

    Loop2_EnableNoclip()

    for i = 1, 50 do
        Loop2_SNOWship(root)

        if Loop2_CheckNetworkOwnership(targetPlayer) or root.AssemblyLinearVelocity.Magnitude > 500 then
            Loop2_CreateSkyVelocity(root)
            ungrab(root)
            break
        end

        task.wait()

        if root.Position.Y <= -12 then
            Loop2_TeleportPlayer(CFrame.new(root.Position + Vector3.new(0, 5, -15)))
        else
            Loop2_TeleportPlayer(CFrame.new(root.Position + Vector3.new(0, -10, -10)))
        end

        humanoid.BreakJointsOnDeath = false
        humanoid:ChangeState(Enum.HumanoidStateType.Dead)
        humanoid.Jump = true
        humanoid.Sit = false
    end

    Loop2_DisableNoclip()
    Loop2_TeleportPlayer(initialPos)
end

local function Loop2_BringPlayer(targetPlayer)
    if not Loop2_IsValidTarget(targetPlayer) then return end

    local root = targetPlayer.Character:FindFirstChild("HumanoidRootPart")
    local humanoid = targetPlayer.Character:FindFirstChildOfClass("Humanoid")
    if not root or not humanoid then return end

    local initialPos = LP.Character.HumanoidRootPart.CFrame
    local myPos = LP.Character.HumanoidRootPart.Position

    Loop2_EnableNoclip()

    for i = 1, 50 do
        Loop2_SNOWship(root)

        if Loop2_CheckNetworkOwnership(targetPlayer) then
            root.CFrame = CFrame.new(myPos + Vector3.new(0, 3, 0))
            ungrab(root)
            break
        end

        task.wait()

        if root.Position.Y <= -12 then
            Loop2_TeleportPlayer(CFrame.new(root.Position + Vector3.new(0, 5, -15)))
        else
            Loop2_TeleportPlayer(CFrame.new(root.Position + Vector3.new(0, -10, -10)))
        end

        humanoid.Jump = true
        humanoid.Sit = false
    end

    Loop2_DisableNoclip()
    Loop2_TeleportPlayer(initialPos)
end

local function Loop2_KickPlayer(targetPlayer)
    if not Loop2_IsValidKickTarget(targetPlayer) then return end

    local root = targetPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not root then return end

    local initialPos = Loop2_GetPlayerCFrame()

    for i = 0, 50 do
        Loop2_EnableNoclip()
        Loop2_SNOWship(root)

        if not Loop2_IsValidKickTarget(targetPlayer) or not Config.Loop2.Kick then
            Loop2_CreateSkyVelocity(root)
            ungrab(root)
            break
        end

        if Loop2_CheckNetworkOwnership(targetPlayer) or root.AssemblyLinearVelocity.Magnitude > 500 then
            Loop2_CreateSkyVelocity(root)
            ungrab(root)
            break
        end

        task.wait()

        if root.Position.Y <= -12 then
            Loop2_TeleportPlayer(CFrame.new(root.Position + Vector3.new(0, 5, -15)))
        else
            Loop2_TeleportPlayer(CFrame.new(root.Position + Vector3.new(0, -10, -10)))
        end
    end

    Loop2_DisableNoclip()
    Loop2_TeleportPlayer(initialPos)
end

local function findFireworkSparklers()
    local toys = {}
    for _, item in ipairs(Workspace:GetDescendants()) do
        if item:IsA("Model") and item.Name == "FireworkSparkler" then
            local alreadyAdded = false
            for _, existingToy in ipairs(toys) do
                if existingToy == item then
                    alreadyAdded = true
                    break
                end
            end
            if not alreadyAdded then
                table.insert(toys, item)
            end
        end
    end
    table.sort(toys, function(a, b) return a.Name < b.Name end)
    return toys
end

local function getPrimaryPart(model)
    if model.PrimaryPart then return model.PrimaryPart end
    local potentialParts = {"Handle", "Main", "Part", "Base", "Sparkler", "Firework"}
    for _, partName in ipairs(potentialParts) do
        local part = model:FindFirstChild(partName)
        if part and part:IsA("BasePart") then return part end
    end
    for _, child in ipairs(model:GetChildren()) do
        if child:IsA("BasePart") then return child end
    end
    return nil
end

local function attachRingPhysics(rec)
    local model = rec.model
    local part = rec.part
    if not model or not part or not part.Parent then return end

    for _, p in ipairs(model:GetDescendants()) do
        if p:IsA("BasePart") then
            pcall(function() p:SetNetworkOwner(LP) end)
            p.CanCollide = false
            p.CanTouch = false
        end
    end

    if not part:FindFirstChild("RingBodyVelocity") then
        local bv = Instance.new("BodyVelocity")
        bv.Name = "RingBodyVelocity"
        bv.MaxForce = Vector3.new(1e8, 1e8, 1e8)
        bv.Velocity = Vector3.new()
        bv.P = 1e6
        bv.Parent = part
    end

    if not part:FindFirstChild("RingBodyGyro") then
        local bg = Instance.new("BodyGyro")
        bg.Name = "RingBodyGyro"
        bg.MaxTorque = Vector3.new(1e8, 1e8, 1e8)
        bg.CFrame = part.CFrame
        bg.P = 1e6
        bg.Parent = part
    end
end

local function detachRingPhysics(rec)
    local model = rec.model
    local part = rec.part
    if not model or not part then return end

    local bv = part:FindFirstChild("RingBodyVelocity")
    if bv then bv:Destroy() end

    local bg = part:FindFirstChild("RingBodyGyro")
    if bg then bg:Destroy() end

    for _, p in ipairs(model:GetDescendants()) do
        if p:IsA("BasePart") then
            p.CanCollide = true
            p.CanTouch = true
            pcall(function() p:SetNetworkOwner(nil) end)
        end
    end
end

local function rescanRing()
    for _, r in ipairs(Config.Ring.RingList) do
        detachRingPhysics(r)
    end
    Config.Ring.RingList = {}

    local foundCount = 0
    for _, d in ipairs(Workspace:GetDescendants()) do
        if foundCount >= Config.Ring.Count then break end

        if d:IsA("Model") and d.Name == "FireworkSparkler" then
            local part = getPrimaryPart(d)
            if part and not part.Anchored then
                local rec = { model = d, part = part }
                table.insert(Config.Ring.RingList, rec)
                foundCount = foundCount + 1
            end
        end
    end

    for i = 1, #Config.Ring.RingList do
        attachRingPhysics(Config.Ring.RingList[i])
    end
    OrionLib:MakeNotification({Name="リング", Content="花火を再スキャンしました。数: "..#Config.Ring.RingList, Time=2})
end

local function startRingLoop()
    if Config.Ring.RingLoopConn then
        Config.Ring.RingLoopConn:Disconnect()
        Config.Ring.RingLoopConn = nil
    end
    Config.Ring.RingTAccum = 0

    Config.Ring.RingLoopConn = RunService.Heartbeat:Connect(function(dt)
        if not Config.Ring.Enabled then return end

        local root = HRP()
        if not root or #Config.Ring.RingList == 0 then return end

        Config.Ring.RingTAccum = Config.Ring.RingTAccum + dt * (Config.Ring.Speed / 10)

        local radius = Config.Ring.Diameter / 2
        local angleIncrement = 360 / #Config.Ring.RingList
        local rootVelocity = root.AssemblyLinearVelocity or root.Velocity or Vector3.new()

        for i, rec in ipairs(Config.Ring.RingList) do
            local part = rec.part
            if not part or not part.Parent then continue end

            local angle = math.rad(i * angleIncrement + Config.Ring.RingTAccum * 50)
            local localPos = Vector3.new(
                radius * math.cos(angle),
                Config.Ring.Height,
                radius * math.sin(angle)
            )
            local targetPos = root.Position + localPos

            local dir = targetPos - part.Position
            local distance = dir.Magnitude
            local bv = part:FindFirstChild("RingBodyVelocity")
            if bv then
                if distance > 0.1 then
                    local moveVelocity = dir.Unit * math.min(3000, distance * 50)
                    bv.Velocity = moveVelocity + rootVelocity
                else
                    bv.Velocity = rootVelocity
                end
            end

            local bg = part:FindFirstChild("RingBodyGyro")
            if bg then
                local lookAtCFrame = CFrame.lookAt(targetPos, root.Position) * CFrame.Angles(0, math.pi, 0)
                bg.CFrame = lookAtCFrame
            end
        end
    end)
end

local function stopRingLoop()
    if Config.Ring.RingLoopConn then

-- TP Walk System
local function updateTPWalk()
    if Config.Player.tpWalkConnection then
        Config.Player.tpWalkConnection:Disconnect()
        Config.Player.tpWalkConnection = nil
    end
    
    if Config.Player.TPWalkEnabled then
        Config.Player.tpWalkConnection = RunService.RenderStepped:Connect(function()
            local character = LP.Character
            if not character then return end
            
            local hrp = character:FindFirstChild("HumanoidRootPart")
            local humanoid = character:FindFirstChild("Humanoid")
            if not hrp or not humanoid then return end
            
            local moveDir = humanoid.MoveDirection
            if moveDir.Magnitude > 0 then
                hrp.CFrame = hrp.CFrame + (moveDir * Config.Player.TPSpeed)
            end
        end)
    end
end

local function toggleTPWalk()
    Config.Player.TPWalkEnabled = not Config.Player.TPWalkEnabled
    updatePlayerMovement()
    
    OrionLib:MakeNotification({
        Name = "TP Walk",
        Content = "TP Walk: " .. (Config.Player.TPWalkEnabled and "ON" or "OFF"),
        Time = 2
    })
end

local function updateInfJump()
    if Config.Player.infJumpConnection then
        Config.Player.infJumpConnection:Disconnect()
        Config.Player.infJumpConnection = nil
    end
    
    if Config.Player.InfJump then
        local character = LP.Character
        local humanoid = character and character:FindFirstChild("Humanoid")
        
        if humanoid then
            humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
        end
        
        Config.Player.infJumpConnection = UserInputService.JumpRequest:Connect(function()
            if Config.Player.InfJump and LP.Character then
                local hum = LP.Character:FindFirstChild("Humanoid")
                if hum then
                    hum:ChangeState(Enum.HumanoidStateType.Jumping)
                end
            end
        end)

local function updatePlayerMovement()
    if Config.Player.speedConnection then Config.Player.speedConnection:Disconnect() end
    if Config.Player.jumpConnection then Config.Player.jumpConnection:Disconnect() end
    if Config.Player.noclipConnection then Config.Player.noclipConnection:Disconnect() end
    
    if Config.Player.SpeedEnabled then
        Config.Player.speedConnection = RunService.Heartbeat:Connect(function()
            if LP.Character and LP.Character:FindFirstChild("Humanoid") then
                LP.Character.Humanoid.WalkSpeed = Config.Player.SpeedValue
            end
        end)
    end
    
    if Config.Player.JumpEnabled then
        Config.Player.jumpConnection = RunService.Heartbeat:Connect(function()
            if LP.Character and LP.Character:FindFirstChild("Humanoid") then
                LP.Character.Humanoid.JumpPower = Config.Player.JumpValue
            end
        end)
    end
    
    if Config.Player.Noclip then
        Config.Player.noclipConnection = RunService.Stepped:Connect(function()
            if LP.Character then
                for _, part in ipairs(LP.Character:GetDescendants()) do
                    if part:IsA("BasePart") then
                        part.CanCollide = false
                    end
                end
            end
        end)
    end
    
    updateTPWalk()
    updateInfJump()
end

local function Snipefunc(root, func, ...)
    if not root or not root.Parent then return end
    local pos = HRP().CFrame
    local args = {...}
    task.spawn(function()
        local parts = {"Head", "Torso", "HumanoidRootPart"}
        for _, p in pairs(parts) do
            local part = LP.Character:FindFirstChild(p)
            if part then part.CanCollide = false end
        end
        local targetPos = root.Position
        HRP().CFrame = CFrame.new(targetPos.X, targetPos.Y - 6, targetPos.Z)
        task.wait(0.1)
        Workspace.CurrentCamera.CFrame = CFrame.lookAt(Workspace.CurrentCamera.CFrame.Position, root.Position)
        for _ = 1, 4 do
            SetNetworkOwner(root, HRP().CFrame)
            task.wait(0.05)
        end
        local look = Workspace.CurrentCamera.CFrame
        task.wait(0.1)
        func(unpack(args))
        Workspace.CurrentCamera.CFrame = look
        task.wait(0.1)
        for _, p in pairs(parts) do
            local part = LP.Character:FindFirstChild(p)
            if part then part.CanCollide = true end
        end
        HRP().CFrame = pos
        Velocity(HRP(), Vector3.zero)
    end)
end

local antiGucciRunning = false
local antiGucciToyName = "CreatureBlobman"
local antiGucciInstance = nil
local antiGucciConnection = nil
local antiGucciOriginalPos = nil
local ANTI_DURATION = 0.5
local SPAWN_POS = Vector3.new(0, 999999999999999, 0)

local function clearAntiGucciRagdoll()
    local _, hrp, hum = HRP(), getHum()
    if hrp and hum then
        pcall(function()
            ragdollRemote:FireServer(hrp, 0)
            hum:ChangeState(Enum.HumanoidStateType.Jumping)
        end)
    end
end

local function executeAntiGucciSequence(child)
    if child.Name ~= antiGucciToyName then return end
    antiGucciInstance = child
    
    local hrp = HRP()
    local hum = getHum()
    if not (hrp and hum) then return end

    local seat = child:WaitForChild("VehicleSeat", 2) or child:FindFirstChildWhichIsA("VehicleSeat", true)
    
    if seat and hum then
        seat:Sit(hum)
        
        local start = tick()
        while tick() - start < ANTI_DURATION and antiGucciRunning do
            pcall(function()
                ragdollRemote:FireServer(hrp, 0)
                hum:ChangeState(Enum.HumanoidStateType.Jumping)
            end)
            RunService.Heartbeat:Wait()
        end
        
        local primary = child.PrimaryPart or child:FindFirstChild("HumanoidRootPart", true) or child:FindFirstChild("Part", true)
        if primary and antiGucciRunning then
            pcall(function()
                if primary.SetNetworkOwner then primary:SetNetworkOwner(LP) end
                sethiddenproperty(primary, "NetworkIsSleeping", false)
            end)
        end

        if antiGucciOriginalPos then
            hrp.CFrame = antiGucciOriginalPos
        end

        task.wait(0.1)
        if seat then
            seat:Destroy()
        end
    end
end

local function spawnAntiGucciProcess()
    if not antiGucciRunning then return end
    
    pcall(function()
        spawnRemote:InvokeServer(
            antiGucciToyName,
            CFrame.new(SPAWN_POS),
            Vector3.new(0, -15.716, 0)
        )
    end)
end

local function startAntiGucciSystem()
    local hrp = HRP()
    if hrp then
        antiGucciOriginalPos = hrp.CFrame
    end

    if antiGucciConnection then antiGucciConnection:Disconnect() end
    
    antiGucciConnection = SpawnedInToys.ChildAdded:Connect(function(child)
        task.spawn(function() executeAntiGucciSequence(child) end)
    end)

    task.spawn(function()
        while antiGucciRunning do
            if not antiGucciInstance or not antiGucciInstance.Parent then
                spawnAntiGucciProcess()
            end
            task.wait(1)
        end
    end)
end

local function toggleAntiGucci(state)

    antiGucciRunning = state
    
    if state then

                hrp:GetPropertyChangedSignal("Velocity"):Connect(function()
                    if Config.Anti.AntiKillV2 and hrp.Velocity.Magnitude > 50 then

local function ExecuteKickAllV2()
    local targets = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            table.insert(targets, p)
        end
    end

    if #targets == 0 then
        OrionLib:MakeNotification({Name="Kick All V2", Content="対象プレイヤーがいません", Time=3})
        return
    end

    local blob = GetBlobman() or SpawnAndSitBlobman()
    if not blob then
        OrionLib:MakeNotification({Name="Kick All V2", Content="Blobmanの召喚に失敗しました", Time=3})
        return
    end

    local myHRP = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
    if not myHRP then return end

    -- 各人にGrabRelease×3
    for _, target in ipairs(targets) do
        local tRoot = target.Character.HumanoidRootPart
        if tRoot then
            myHRP.CFrame = tRoot.CFrame * CFrame.new(0, 0, -1.8)
            task.wait(0.05)

            for i = 1, 3 do
                pcall(function()
                    blob.BlobmanSeatAndOwnerScript.CreatureGrab:FireServer(
                        blob.LeftDetector, tRoot, blob.LeftDetector.LeftWeld)
                end)
                pcall(function()
                    blob.BlobmanSeatAndOwnerScript.CreatureRelease:FireServer(
                        blob.LeftDetector.LeftWeld)
                end)
                task.wait(0.08)
            end
        end
    end

    -- 中心移動
    myHRP.CFrame = CFrame.new(0, 105, 0)
    task.wait(0.15)

    -- 円形配置
    local radius = 17
    local center = CFrame.new(0, 110, 0)
    for i, plr in ipairs(targets) do
        local ang = math.rad((i-1) * (360 / #targets))
        local x = radius * math.cos(ang)
        local z = radius * math.sin(ang)
        pcall(function()
            plr.Character.HumanoidRootPart.CFrame = CFrame.new(center.Position + Vector3.new(x, 0, z))
        end)
    end
    task.wait(0.12)

    -- Aura風 ×2
    local GrabEvents = ReplicatedStorage:FindFirstChild("GrabEvents")
    local SetNet = GrabEvents and GrabEvents:FindFirstChild("SetNetworkOwner")
    local DestroyLine = GrabEvents and GrabEvents:FindFirstChild("DestroyGrabLine")

    if SetNet and DestroyLine then
        for _ = 1, 2 do
            for _, plr in ipairs(targets) do
                local root = plr.Character and plr.Character:FindFirstChild("HumanoidRootPart")
                if root then
                    pcall(function()
                        SetNet:FireServer(root, myHRP.CFrame)
                        DestroyLine:FireServer(root)
                    end)
                end
            end
            task.wait(0.09)
        end
    end

    task.wait(0.35)

    -- 両手キック
    for _, plr in ipairs(targets) do
        local root = plr.Character and plr.Character:FindFirstChild("HumanoidRootPart")
        if not root then continue end
        pcall(function()
            blob.BlobmanSeatAndOwnerScript.CreatureGrab:FireServer(
                blob.LeftDetector, root, blob.LeftDetector.LeftWeld)
        end)
        pcall(function()
            blob.BlobmanSeatAndOwnerScript.CreatureGrab:FireServer(
                blob.RightDetector, root, blob.RightDetector.RightWeld)
        end)
    end

    OrionLib:MakeNotification({Name="Kick All V2", Content=#targets.."人をキックしました", Time=4})
end

local function updateESP()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and p.Character then
            local c = p.Character
            local h = c:FindFirstChild("Highlight")
            if Config.ESP.Enabled then
                if not h then
                    highlightCharacter(c)
                else
                    h.FillColor = Config.ESP.HighlightColor
                    h.FillTransparency = Config.ESP.HighlightTransparency
                end
            elseif h then
                h:Destroy()
            end
        end
    end
end

-- Wing Functions
local WingToys = {}
local WingRowPoints = {}
local WingAssignedToys = {}
local WingLoopConn = nil
local WingTime = 0

local function findFireworkSparklersWing()
    local sparklers = {}
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj.Name == "FireworkSparkler" and obj:IsA("Model") then
            table.insert(sparklers, obj)
        end
    end
    return sparklers
end

local function getPrimaryPartWing(model)
    return model.PrimaryPart or model:FindFirstChildWhichIsA("BasePart")
end

local function createWingRowPoints(count)
    local points = {}
    
    if count == 0 then return points end
    
    local totalWidth = (count - 1) * Config.Wing.spacing
    local startX = -totalWidth / 2
    
    for i = 1, count do
        local x = startX + (i - 1) * Config.Wing.spacing
        local part = Instance.new("Part")
        part.CanCollide = false
        part.Anchored = true
        part.Transparency = 1
        part.Size = Vector3.new(4, 1, 4)
        part.Parent = Workspace
        
        points[i] = {
            offsetX = x,
            part = part,
            assignedToy = nil,
        }
    end
    
    return points
end

local function attachWingPhysics(part)
    if not part then return nil, nil end
    
    local existingBG = part:FindFirstChildOfClass("BodyGyro")
    local existingBP = part:FindFirstChildOfClass("BodyPosition")
    
    if existingBG and existingBP then 
        return existingBG, existingBP
    end
    
    if existingBG then existingBG:Destroy() end
    if existingBP then existingBP:Destroy() end
    
    local BP = Instance.new("BodyPosition")  
    local BG = Instance.new("BodyGyro")  
    
    BP.P = 15000  
    BP.D = 200  
    BP.MaxForce = Vector3.new(1, 1, 1) * 1e10  
    BP.Parent = part  
    
    BG.P = 15000  
    BG.D = 200  
    BG.MaxTorque = Vector3.new(1, 1, 1) * 1e10  
    BG.Parent = part  
    
    return BG, BP
end

local function assignWingToysToPoints()
    WingAssignedToys = {}
    local distanceGroups = {}
    
    for i, point in ipairs(WingRowPoints) do
        local absDistance = math.abs(point.offsetX)
        
        if not distanceGroups[absDistance] then
            distanceGroups[absDistance] = {}
        end
        table.insert(distanceGroups[absDistance], i)
    end
    
    local sortedDistances = {}
    for distance, _ in pairs(distanceGroups) do
        table.insert(sortedDistances, distance)
    end
    table.sort(sortedDistances)
    
    for rank, distance in ipairs(sortedDistances) do
        for _, pointIndex in ipairs(distanceGroups[distance]) do
            WingRowPoints[pointIndex].distanceRank = rank
        end
    end
    
    for i = 1, math.min(#WingToys, #WingRowPoints) do
        local toy = WingToys[i]
        if toy and toy:IsA("Model") and toy.Name == "FireworkSparkler" then
            local primaryPart = getPrimaryPartWing(toy)
            
            if primaryPart then  
                for _, child in ipairs(toy:GetChildren()) do  
                    if child:IsA("BasePart") then  
                        child.CanCollide = false
                        child.CanTouch = false
                        child.Anchored = false
                    end  
                end
                
                local BG, BP = attachWingPhysics(primaryPart)  
                local toyTable = {  
                    BG = BG,  
                    BP = BP,  
                    Pallet = primaryPart,
                    Model = toy,
                    RowIndex = i,
                    offsetX = WingRowPoints[i].offsetX,
                    distanceRank = WingRowPoints[i].distanceRank
                }  
                
                WingRowPoints[i].assignedToy = toyTable
                table.insert(WingAssignedToys, toyTable)
            end  
        end
    end
    
    return WingAssignedToys
end

local function startWingLoop()
    if WingLoopConn then
        WingLoopConn:Disconnect()
        WingLoopConn = nil
    end
    
    WingTime = 0
    
    WingLoopConn = RunService.RenderStepped:Connect(function(dt)
        if not Config.Wing.Enabled or not LP.Character then
            return
        end
        
        local humanoidRootPart = LP.Character:FindFirstChild("HumanoidRootPart")
        local torso = LP.Character:FindFirstChild("Torso") or LP.Character:FindFirstChild("UpperTorso")
        
        if not humanoidRootPart or not torso then
            return
        end
        
        WingTime += dt * Config.Wing.waveSpeed
        
        local charCFrame = humanoidRootPart.CFrame
        local rightVector = charCFrame.RightVector
        local lookVector = charCFrame.LookVector
        
        local backVector = -lookVector
        
        local basePosition = torso.Position + 
                             Vector3.new(0, Config.Wing.heightOffset, 0) + 
                             (backVector * Config.Wing.backwardOffset)
        
        for i, point in ipairs(WingRowPoints) do
            if point.assignedToy and point.assignedToy.BP and point.assignedToy.BG then
                local toy = point.assignedToy
                
                local targetPosition = basePosition + (rightVector * toy.offsetX)
                
                local amplitude = Config.Wing.baseAmplitude * toy.distanceRank
                local waveMovement = math.sin(WingTime) * amplitude
                local finalPosition = targetPosition + Vector3.new(0, waveMovement, 0)
                
                if point.part then
                    point.part.Position = finalPosition
                end
                
                toy.BP.Position = finalPosition
                
                local backYRotation = math.atan2(-lookVector.X, -lookVector.Z)
                local baseCFrame = CFrame.new(finalPosition) * CFrame.Angles(0, backYRotation, 0)
                local tiltedCFrame = baseCFrame * CFrame.Angles(math.rad(-Config.Wing.tiltAngle), 0, 0)
                
                local currentCFrame = toy.BG.CFrame
                local interpolatedCFrame = currentCFrame:Lerp(tiltedCFrame, 0.3)
                
                toy.BG.CFrame = interpolatedCFrame
            end
        end
    end)
end

local function stopWingLoop()
    if WingLoopConn then
        WingLoopConn:Disconnect()
        WingLoopConn = nil
    end
    
    for _, point in ipairs(WingRowPoints) do
        if point.part then
            point.part:Destroy()
        end
        if point.assignedToy then
            if point.assignedToy.BG then
                point.assignedToy.BG:Destroy()
            end
            if point.assignedToy.BP then
                point.assignedToy.BP:Destroy()
            end
        end
    end
    
    WingRowPoints = {}
    WingAssignedToys = {}
end

local function toggleWing(state)
    Config.Wing.Enabled = state
    if state then
        WingToys = findFireworkSparklersWing()
        WingRowPoints = createWingRowPoints(math.min(#WingToys, Config.Wing.maxSparklers))
        WingAssignedToys = assignWingToysToPoints()
        startWingLoop()
        
        OrionLib:MakeNotification({
            Name = "ウィング起動",
            Content = "花火数: " .. #WingAssignedToys .. "本 (背中側)",
            Image = "rbxassetid://4483362458",
            Time = 3
        })
    else
        stopWingLoop()
        OrionLib:MakeNotification({
            Name = "ウィング停止",
            Content = "Wingの配置を解除しました",
            Image = "rbxassetid://4483362458",
            Time = 2
        })
    end
end

local function executeKick()
    if isKickExecuting then
        OrionLib:MakeNotification({Name = "キック", Content = "既に実行中です", Time = 3})
        return
    end
    
    if not selectedPlayerName then
        OrionLib:MakeNotification({Name = "エラー", Content = "プレイヤーが選択されていません", Time = 3})
        return
    end
    
    local target = getPlayerByDisplayName(selectedPlayerName)
    if not target then
        OrionLib:MakeNotification({Name = "エラー", Content = "プレイヤーが見つかりません", Time = 3})
        return
    end
    
    local targetChar = target.Character
    if not targetChar then
        OrionLib:MakeNotification({Name = "エラー", Content = "ターゲットのキャラクターが見つかりません", Time = 3})
        return
    end
    
    local targetRoot = targetChar:FindFirstChild("HumanoidRootPart")
    if not targetRoot then
        OrionLib:MakeNotification({Name = "エラー", Content = "ターゲットのHumanoidRootPartが見つかりません", Time = 3})
        return
    end
    
    isKickExecuting = true
    
    task.spawn(function()
        local success = false
        
        pcall(function()
            local blob = getBlobmanForKick()
            if not blob then
                blob = spawnBlobmanForKick()
                if not blob then
                    OrionLib:MakeNotification({Name = "エラー", Content = "Blobmanの召喚に失敗しました", Time = 3})
                    return
                end
            end
            
            if not sitOnBlobmanForKick(blob) then
                OrionLib:MakeNotification({Name = "エラー", Content = "Blobmanに乗れませんでした", Time = 3})
                return
            end
            
            local myRoot = HRP()
            if myRoot and targetRoot then
                local originalPos = myRoot.CFrame
                
                myRoot.CFrame = targetRoot.CFrame
                task.wait(0.07)
                
                blobKickForKick(blob, targetRoot, "Left")
                task.wait(0.25)
                
                myRoot.CFrame = originalPos
            end
            
            destroyBlobmanForKick()
            success = true
        end)
        
        if success then
            OrionLib:MakeNotification({Name = "キック", Content = target.DisplayName .. " をキックしました！", Time = 4})
        end
        
        isKickExecuting = false
    end)
end

            })
        end
    else
        OrionLib:MakeNotification({
            Name = "テレポート補助",
            Content = "近くのプレイヤーが見つかりません",
            Time = 3
        })
    end
end

                        MoveTo(part, CFrame.new(4096, -75, 4096))

        local hrp = HRP()
        if hrp and hrp.Position.Y < -80 then
            hrp.CFrame = CFrame.new(0, 10, 0)
        end
    end


    if Config.MiscL.DefenseTimer >= 0.1 then
        if Config.Anti.AntiExplode then
            local myRoot = HRP()

            if myRoot then
                for _, part in ipairs(workspace:GetChildren()) do
                    if part.Name == "Part" then
                        local distance = (part.Position - myRoot.Position).Magnitude

                        if distance < 4 then
                            myRoot.Anchored = true

                            Struggle()
                            task.wait()
                            myRoot.Anchored = false

RunService.RenderStepped:Connect(function()
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LP and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            local char = player.Character
            local hrp = char:FindFirstChild("HumanoidRootPart")
            
            if Config.ESP.SkeletonEnabled then
                local bones = {
                    Head = GetBone(char, "Head"),
                    Torso = GetBone(char, "UpperTorso") or GetBone(char, "Torso"),
                    LeftArm = GetBone(char, "LeftUpperArm") or GetBone(char, "Left Arm"),
                    RightArm = GetBone(char, "RightUpperArm") or GetBone(char, "Right Arm"),
                    LeftLeg = GetBone(char, "LeftUpperLeg") or GetBone(char, "Left Leg"),
                    RightLeg = GetBone(char, "RightUpperLeg") or GetBone(char, "Right Leg")
                }
                local lines = GetLineTable(player)
                local function drawLine(l, a, b)
                    if a and b then
                        local af, av = Camera:WorldToViewportPoint(a.Position)
                        local bf, bv = Camera:WorldToViewportPoint(b.Position)
                        if av and bv then
                            l.From = Vector2.new(af.X, af.Y)
                            l.To = Vector2.new(bf.X, bf.Y)
                            l.Color = Config.ESP.SkeletonColor
                            l.Thickness = Config.ESP.SkeletonThickness
                            l.Visible = true
                        else
                            l.Visible = false
                        end
                    else 
                        l.Visible = false 
                    end
                end
                drawLine(lines.Head, bones.Head, bones.Torso)
                drawLine(lines.Neck, bones.Head, bones.Torso)
                drawLine(lines.Spine, bones.Torso, hrp)
                drawLine(lines.LeftArm, bones.Torso, bones.LeftArm)
                drawLine(lines.RightArm, bones.Torso, bones.RightArm)
                drawLine(lines.LeftLeg, hrp, bones.LeftLeg)
                drawLine(lines.RightLeg, hrp, bones.RightLeg)
            end
            
            if Config.ESP.TracerEnabled then
                if not TracerLines[player] then createTracer(player) end
                local t = TracerLines[player]
                local pos, vis = Camera:WorldToViewportPoint(hrp.Position)
                t.Visible = vis
                if vis then
                    t.From = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y)
                    t.To = Vector2.new(pos.X, pos.Y)
                    t.Color = getRainbowColor(player.UserId % 10)
                end
            elseif TracerLines[player] then
                TracerLines[player].Visible = false
            end
            
            if not nameESPObjects[player] then
                local t = Drawing.new("Text")
                t.Center = true
                t.Outline = true
                t.Font = 2
                nameESPObjects[player] = t
            end
            local text = nameESPObjects[player]
            local head = char:FindFirstChild("Head")
            if Config.ESP.NameEnabled and head then
                local pos, vis = Camera:WorldToViewportPoint(head.Position + Vector3.new(0, 1.5, 0))
                text.Text = player.Name
                text.Position = Vector2.new(pos.X, pos.Y)
                text.Color = Config.ESP.NameColor
                text.Size = Config.ESP.NameSize
                text.Visible = vis
            else
                text.Visible = false
            end
        end
    end
    
    for player, text in pairs(nameESPObjects) do
        if not Players:FindFirstChild(player.Name) then
            text:Remove()
            nameESPObjects[player] = nil
        end
    end
end)

Workspace.ChildAdded:Connect(function(v)
    if v.Name == "GrabParts" and v:IsA("Model") then
        local GrabPart = v:FindFirstChild("GrabPart")
        if not GrabPart then return end
        
        local WeldConstraint = GrabPart:FindFirstChild("WeldConstraint")
        if not WeldConstraint or not WeldConstraint.Part1 then return end
        
        local target = WeldConstraint.Part1
        
        if Config.Grab.SuperStrength then
            v:GetPropertyChangedSignal("Parent"):Connect(function()
                if not v.Parent and target and target.Parent then
                    local throwVel = Instance.new("BodyVelocity")
                    throwVel.MaxForce = Vector3.new(1e8, 1e8, 1e8)
                    throwVel.Velocity = Workspace.CurrentCamera.CFrame.LookVector * Config.Grab.StrengthPower
                    throwVel.Parent = target
                    Debris:AddItem(throwVel, 0.1)
                end
            end)
        end
        
        task.spawn(function()
            task.wait(0.1)
            
            if Config.Grab.VoidGrab then
                SetNetworkOwner(target)
                Velocity(target, Vector3.new(0, 10000, 0))
            end
            
            if Config.Grab.KillGrab then
                SetNetworkOwner(target)
                MoveTo(target, CFrame.new(4096, -75, 4096))
                Velocity(target, Vector3.new(0, -1000, 0))
            end
            
            if Config.Grab.KickGrab then
                local player = Players:GetPlayerFromCharacter(target.Parent)
                if player then
                    SetNetworkOwner(target)
                    MoveTo(target, CFrame.new(25e25, 25e25, 25e25))
                    task.wait(0.5)
                    ungrab(target)
                end
            end
            
            if Config.Grab.AnchorGrab then
                SetNetworkOwner(target)
                local pos = target.CFrame
                for _ = 1, 2 do
                    SetNetworkOwner(target)
                    local bp = Instance.new("BodyPosition")
                    bp.Position = pos.Position
                    bp.MaxForce = Vector3.new(1e8, 1e8, 1e8)
                    bp.Parent = target
                    
                    local bg = Instance.new("BodyGyro")
                    bg.CFrame = pos
                    bg.MaxTorque = Vector3.new(1e8, 1e8, 1e8)
                    bg.Parent = target
                    
                    task.wait(0.5)
                end
            end
            
            if Config.Grab.LagEnabled then
                lag(Config.Grab.LagAmount)
            end
        end)
    end
end)

    for _, p in Workspace.PlotItems:GetChildren() do
        local m = p:FindFirstChild("CreatureBlobman")
        if m and m:FindFirstChild("PlayerValue") and m.PlayerValue.Value == LP.Name then
            return m
        end
    end
    return nil
end

local function spawnBlobmanForKickAll()
    local root = HRP()
    if not root then return nil end
    pcall(function()
        ReplicatedStorage.MenuToys.SpawnToyRemoteFunction:InvokeServer("CreatureBlobman", root.CFrame, Vector3.zero)
    end)
    task.wait(1.0)
    return getBlobmanForKickAll()
end

local function destroyBlobmanForKickAll()
    local blob = getBlobmanForKickAll()
    if blob then
        pcall(function()
            ReplicatedStorage.MenuToys.DestroyToy:FireServer(blob)
        end)
    end
end

local function sitOnBlobmanForKickAll(blob)
    if not blob or not blob:FindFirstChild("VehicleSeat") then return false end
    local seat = blob.VehicleSeat
    local hum = getHum()
    if hum and not hum.Sit then
        seat:Sit(hum)
        task.wait(0.6)
    end
    return hum and hum.Sit and hum.SeatPart == seat
end

local function blobGrabForKickAll(blob, targetPart, side)
    if not blob then return end
    local detector = blob:FindFirstChild(side .. "Detector")
    if not detector then return end
    local weld = detector:FindFirstChild(side .. "Weld")
    if not weld then return end
    
    local script = blob:FindFirstChild("BlobmanSeatAndOwnerScript", true)
    if script then
        local remote = script:FindFirstChild("CreatureGrab")
        if remote then
            pcall(function() remote:FireServer(detector, targetPart, weld) end)
        end
    end
end

    if not blob then return end
    local detector = blob:FindFirstChild(side .. "Detector")
    if not detector then return end
    
    local script = blob:FindFirstChild("BlobmanSeatAndOwnerScript", true)
    if script then
        local remote = script:FindFirstChild("CreatureDrop")
        if remote then
            pcall(function() remote:FireServer(detector, targetPart) end)
        end
    end
end

local function blobKickForKickAll(blob, targetPart, side)
    local myRoot = HRP()
    if not myRoot then return end
    
    blobGrabForKickAll(blob, myRoot, side)
    task.wait(0.08)
    
    SetNetworkOwner(targetPart)
    task.wait(0.08)
    
    targetPart.CFrame = targetPart.CFrame + Vector3.new(0, 16, 0)
    task.wait(0.08)
    
    ungrab(targetPart)
    task.wait(0.08)
    
    blobGrabForKickAll(blob, targetPart, side)
    task.wait(0.08)
    
    blobDropForKickAll(blob, targetPart, side)
    task.wait(0.08)
    
    ungrab(targetPart)
end

local function executeKickAll()
    if isKickAllExecuting then
        OrionLib:MakeNotification({
            Name = "Kick All",
            Content = "既に実行中です",
            Time = 3
        })
        return
    end
    
    isKickAllExecuting = true
    
    task.spawn(function()
        local success = false
        
        pcall(function()
            local blob = getBlobmanForKickAll()
            if not blob then
                OrionLib:MakeNotification({
                    Name = "Kick All",
                    Content = "Blobmanを召喚中...",
                    Time = 2
                })
                blob = spawnBlobmanForKickAll()
                if not blob then
                    OrionLib:MakeNotification({
                        Name = "Kick All",
                        Content = "Blobmanの召喚に失敗しました",
                        Time = 3
                    })
                    return
                end
            end
            
            if not sitOnBlobmanForKickAll(blob) then
                OrionLib:MakeNotification({
                    Name = "Kick All",
                    Content = "Blobmanに乗れませんでした",
                    Time = 3
                })
                return
            end
            
            local targets = {}
            for _, player in ipairs(Players:GetPlayers()) do
                if player ~= LP and player.Character then
                    local rootPart = player.Character:FindFirstChild("HumanoidRootPart")
                    if rootPart then
                        if rootPart.Position.Y > -25 then
                            table.insert(targets, {
                                player = player,
                                root = rootPart
                            })
                        end
                    end
                end
            end
            
            local blobPos = blob.VehicleSeat.Position
            table.sort(targets, function(a, b)
                return (a.root.Position - blobPos).Magnitude < (b.root.Position - blobPos).Magnitude
            end)
            
            OrionLib:MakeNotification({
                Name = "Kick All",
                Content = string.format("対象プレイヤー: %d人", #targets),
                Time = 3
            })
            
            for _, target in ipairs(targets) do
                if not sitOnBlobmanForKickAll(blob) then
                    break
                end
                
                local myRoot = HRP()
                if myRoot and target.root then
                    local originalPos = myRoot.CFrame
                    
                    myRoot.CFrame = target.root.CFrame
                    task.wait(0.1)
                    
                    blobKickForKickAll(blob, target.root, "Left")
                    task.wait(0.3)
                    
                    myRoot.CFrame = originalPos
                    task.wait(0.1)
                end
            end
            
            destroyBlobmanForKickAll()
            
            success = true
        end)
        
        if success then
            OrionLib:MakeNotification({
                Name = "Kick All",
                Content = "実行完了しました！",
                Time = 4
            })
        end
        
        isKickAllExecuting = false
    end)
end

                    local playerTorso = player.Character:FindFirstChild("Torso") or player.Character:FindFirstChild("HumanoidRootPart")

                        local distance = (playerTorso.Position - center.Position).Magnitude

                end
            end)

local function executeBarrierBreak()
    local player = LP
    local playerName = player.Name
    local plotItems = Workspace:FindFirstChild("PlotItems")
    
    if plotItems then
        local playersInPlots = plotItems:FindFirstChild("PlayersInPlots")
        if playersInPlots and playersInPlots:FindFirstChild(playerName) then
            OrionLib:MakeNotification({Name = "エラー", Content = "家の中に居ないでください", Time = 3})
            return false
        end
    end
    
    local success, err = pcall(function()
        if not player.Character or not player.Character:FindFirstChild("HumanoidRootPart") then
            error("キャラクターが見つかりません")
        end
        
        local humanoid = player.Character:FindFirstChildOfClass("Humanoid")
        if not humanoid then error("Humanoidが見つかりません") end
        
        local originalWalkSpeed = humanoid.WalkSpeed
        local originalPosition = player.Character.HumanoidRootPart.CFrame
        humanoid.WalkSpeed = 0
        
        if not ReplicatedStorage:FindFirstChild("MenuToys") then
            humanoid.WalkSpeed = originalWalkSpeed
            error("MenuToysが見つかりません")
        end
        
        ReplicatedStorage.MenuToys.SpawnToyRemoteFunction:InvokeServer(
            "InstrumentWoodwindOcarina",
            CFrame.new(184.148834, -5.54824972, 498.136749),
            Vector3.new(0, 34, 0)
        )
        task.wait(0.4)
        
        local toyFolder = Workspace:FindFirstChild(player.Name .. "SpawnedInToys")

RunService.RenderStepped:Connect(function(dt)
    local s = Config.SilentAim
    if not s.Enabled or not s.CameraClone then 
        if s.CameraClone then Camera.CFrame = s.CameraClone.CFrame end
        return 
    end

    if s.FOVCircle and s.ShowFOV then
        s.RainbowTimer = math.fmod(s.RainbowTimer + dt * 100, 360)
        s.FOVCircle.Position = UserInputService:GetMouseLocation()
        s.FOVCircle.Color = s.FOVRainbow and Internal_HsvToRgb(s.RainbowTimer, 1, 1) or s.FOVColor
    end

    local TargetCFrame = s.CameraClone.CFrame
    s.TargetHRP = nil 

    if not Workspace:FindFirstChild("GrabParts") then
        local Pos = s.CameraClone.CFrame.Position + (s.CameraClone.CFrame.LookVector * 10)
        local Dist = s.Reach
        local Best = nil

        for HRP, _ in pairs(s.HRPs) do
            if HRP.Parent and HRP.Parent:FindFirstChild("Humanoid") and HRP.Parent.Humanoid.Health > 0 then
                local Mag = (HRP.Position - Pos).Magnitude
                if Mag < Dist then
                    Dist = Mag
                    Best = HRP
                end
            end
        end

        if Best then
            s.TargetHRP = Best 
            if TargetCFrame.LookVector:Dot((Best.Position - TargetCFrame.Position).Unit) > 0 then
                local Head = Best.Parent:FindFirstChild("Head") or Best 
                local Screen3D, _ = s.CameraClone:WorldToScreenPoint(Head.Position)
                local Screen2D = Vector2.new(Screen3D.X, Screen3D.Y)
                
                if (Screen2D - (Camera.ViewportSize / 2)).Magnitude <= s.FOVSize then
                    TargetCFrame = CFrame.new(TargetCFrame.Position, Head.Position)
                else
                    s.TargetHRP = nil 
                end
            end
        end
    end
    Camera.CFrame = TargetCFrame
end)

    local s = Config.SilentAim
    if d:IsA("BasePart") and d.Name == "HumanoidRootPart" then
        local p = Players:GetPlayerFromCharacter(d.Parent)
        if p and p ~= LP then s.HRPs[d] = isAdded or nil end
    end
end
