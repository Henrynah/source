-- 共通便利関数
local function GetAllTargets()
    local targets = {}
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            table.insert(targets, player)
        end
    end
    return targets
end

local function GetMyRoot()
    local character = LocalPlayer.Character
    return character and character:FindFirstChild("HumanoidRootPart")
end

local function ResetTargets()
    for id, status in pairs(playerStatus) do
        if status == "Targeting" then
            playerStatus[id] = nil
        end
    end
end

local function GetPlayerNamesList()
    local list = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then
            table.insert(list, p.Name .. " [" .. tostring(p.UserId) .. "]")
        end
    end
    return list
end

local function GetTargetPlayerFromFormat(selectedStr)
    if not selectedStr or selectedStr == "" then return nil end
    local name = string.gsub(selectedStr, "%s*%[%d+%]$", "")
    return Players:FindFirstChild(name)
end

local function RefreshDropdown()
    if playerDropdown then
        playerDropdown:Refresh(GetPlayerNamesList(), true)
    end
end

-- 石の斧を装備（スロット置換）させる処理の関数化
local function EquipStoneAxe()
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    local character = LocalPlayer.Character
    if not character then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local axe = (backpack and backpack:FindFirstChild("StoneAxe")) or character:FindFirstChild("StoneAxe")
    if not axe then
        axe = Instance.new("Tool")
        axe.Name = "StoneAxe"
        local handle = Instance.new("Part", axe)
        handle.Name = "Handle"
        handle.Size = Vector3.new(1, 1, 1)
        axe.Parent = backpack
    end
    if humanoid and axe then
        humanoid:EquipTool(axe)
    end
end

local function KickAll()
    if isActive then return end
    isActive = true
    local allPlayers = GetAllTargets()
    if #allPlayers == 0 then isActive = false return end
    for _, targetPlayer in ipairs(allPlayers) do
        playerStatus[targetPlayer.UserId] = "Targeting"
    end
    local rootPart = GetMyRoot()
    if rootPart then
        local spawnPos = rootPart.CFrame * CFrame.new(0, 0, -5)
        pcall(function()
            ReplicatedStorage.MenuToys.SpawnToyRemoteFunction:InvokeServer("CreatureBlobman", spawnPos, Vector3.new(0, 127, 0))
        end)
    end
    task.wait(0.5)
    local toyFolder = workspace:FindFirstChild(LocalPlayer.Name .. "SpawnedInToys")
    currentBlob = toyFolder and toyFolder:FindFirstChild("CreatureBlobman")
    if not currentBlob then
        ResetTargets()
        isActive = false
        return
    end
    local vehicleSeat = currentBlob:FindFirstChild("VehicleSeat")
    if vehicleSeat and LocalPlayer.Character then
        local humanoid = LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
        if humanoid then
            pcall(function() vehicleSeat:Sit(humanoid) end)
        end
    end
    task.wait(0.3)
    local myRoot = GetMyRoot()
    if not myRoot then
        ResetTargets()
        isActive = false
        return
    end
    for _, targetPlayer in ipairs(allPlayers) do
        local targetCharacter = targetPlayer.Character
        local targetRoot = targetCharacter and targetCharacter:FindFirstChild("HumanoidRootPart")
        if targetRoot and myRoot.Parent then
            myRoot.CFrame = targetRoot.CFrame
            myRoot.AssemblyAngularVelocity = Vector3.zero
            task.wait(0.025)
            if (myRoot.Position - targetRoot.Position).Magnitude <= 35 then
                for i = 1, 8 do
                    if not targetRoot.Parent or not currentBlob.Parent then break end
                    pcall(function()
                        local scriptRef = currentBlob.BlobmanSeatAndOwnerScript
                        scriptRef.CreatureGrab:FireServer(currentBlob.LeftDetector, targetRoot, currentBlob.LeftDetector.LeftWeld)
                        scriptRef.CreatureRelease:FireServer(currentBlob.LeftDetector.LeftWeld)
                    end)
                    task.wait()
                end
            end
        end
    end
    if myRoot.Parent then
        myRoot.CFrame = CFrame.new(0, 100, 0)
        myRoot.AssemblyLinearVelocity = Vector3.zero
    end
    task.wait(0.1)
    for _, part in ipairs(currentBlob:GetDescendants()) do
        if part:IsA("BasePart") then
            pcall(function() part.Anchored = true end)
        end
    end
    task.wait(0.1)
    for i, targetPlayer in ipairs(allPlayers) do
        local targetCharacter = targetPlayer.Character
        local targetRoot = targetCharacter and targetCharacter:FindFirstChild("HumanoidRootPart")
        if targetRoot then
            local angle = math.rad((i - 1) * (360 / #allPlayers))
            pcall(function()
                targetRoot.CFrame = CFrame.new(15 * math.cos(angle), 110, 15 * math.sin(angle))
            end)
        end
    end
    task.wait(0.1)
    for _ = 1, 2 do
        for _, targetPlayer in ipairs(allPlayers) do
            local targetRoot = targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart")
            if targetRoot then
                task.spawn(pcall, function()
                    ReplicatedStorage.GrabEvents.SetNetworkOwner:FireServer(targetRoot, CFrame.new(targetRoot.Position))
                    ReplicatedStorage.GrabEvents.DestroyGrabLine:FireServer(targetRoot)
                end)
            end
        end
        task.wait(0.1)
    end
    task.wait(0.3)
    for _, targetPlayer in ipairs(allPlayers) do
        local targetRoot = targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart")
        if targetRoot and currentBlob.Parent then
            task.spawn(pcall, function()
                local scriptRef = currentBlob.BlobmanSeatAndOwnerScript
                scriptRef.CreatureGrab:FireServer(currentBlob.LeftDetector, targetRoot, currentBlob.LeftDetector.LeftWeld)
                scriptRef.CreatureGrab:FireServer(currentBlob.RightDetector, targetRoot, currentBlob.RightDetector.RightWeld)
            end)
        end
    end
    task.wait(0.1)
    for _, targetPlayer in ipairs(allPlayers) do
        if targetPlayer and targetPlayer.Parent == Players then
            playerStatus[targetPlayer.UserId] = "Kicked"
        end
    end
    if currentBlob and currentBlob.Parent then
        for _, part in ipairs(currentBlob:GetDescendants()) do
            if part:IsA("BasePart") then
                pcall(function() part.Anchored = false end)
            end
        end
    end
    task.wait(1)
    isActive = false
end

RunService.Stepped:Connect(function()
    local myRoot = GetMyRoot()
    local myChar = LocalPlayer.Character
    if noclip and myChar then
        for _, part in ipairs(myChar:GetDescendants()) do
            if part:IsA("BasePart") then
                part.CanCollide = false
            end
        end
    end
    if loopFollowTarget and loopFollowTarget.Character and myRoot then
        local tRoot = loopFollowTarget.Character:FindFirstChild("HumanoidRootPart")
        if tRoot then
            myRoot.CFrame = tRoot.CFrame * CFrame.new(0, 0, 3)
        end
    end
    for _, p in ipairs(Players:GetPlayers()) do
        if loopFlingTargets[p.UserId] and p.Character and myRoot then
            local tRoot = p.Character:FindFirstChild("HumanoidRootPart")
            if tRoot then
                myRoot.CFrame = tRoot.CFrame
                myRoot.AssemblyLinearVelocity = Vector3.new(99999, 99999, 99999)
            end
        end
    end
    if hitBoxEnabled then
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character then
                local tRoot = p.Character:FindFirstChild("HumanoidRootPart")
                if tRoot then
                    tRoot.Size = Vector3.new(hitBoxSize, hitBoxSize, hitBoxSize)
                    tRoot.Transparency = 0.7
                    tRoot.Color = Color3.fromRGB(255, 0, 0)
                    tRoot.CanCollide = false
                end
            end
        end
    end
    if spiderMode and myChar and myRoot then
        local ray = Ray.new(myRoot.Position, myRoot.CFrame.LookVector * 3)
        local part = workspace:FindPartOnRay(ray, myChar)
        if part then
            myRoot.Velocity = Vector3.new(myRoot.Velocity.X, 30, myRoot.Velocity.Z)
        end
    end
    -- 🛡【各種カウンター防御】
    if myChar and myRoot then
        for _, obj in ipairs(myChar:GetDescendants()) do
            if obj:IsA("Weld") or obj:IsA("Snap") or obj:IsA("ManualWeld") or obj.Name:lower():find("weld") then
                local attacker = nil
                for _, p in ipairs(Players:GetPlayers()) do
                    if p ~= LocalPlayer and p.Character and (obj:IsDescendantOf(p.Character) or (obj.Part0 and obj.Part0:IsDescendantOf(p.Character)) or (obj.Part1 and obj.Part1:IsDescendantOf(p.Character))) then
                        attacker = p
                        break
                    end
                end
                if antiFlyEnabled and attacker and attacker.Character then
                    local aRoot = attacker.Character:FindFirstChild("HumanoidRootPart")
                    if aRoot then
                        aRoot.AssemblyLinearVelocity = Vector3.new(500000, 500000, 500000)
                        aRoot.AssemblyAngularVelocity = Vector3.new(500000, 500000, 500000)
                    end
                end
                if antiAttackerEnabled and attacker and attacker.Character then
                    local aRoot = attacker.Character:FindFirstChild("HumanoidRootPart")
                    if aRoot and currentBlob and currentBlob.Parent then
                        pcall(function()
                            currentBlob.BlobmanSeatAndOwnerScript.CreatureGrab:FireServer(currentBlob.LeftDetector, aRoot, currentBlob.LeftDetector.LeftWeld)
                        end)
                    end
                end
                if antiLockEnabled and attacker and attacker.Character then
                    for _, p in ipairs(attacker.Character:GetDescendants()) do
                        if p:IsA("BasePart") then
                            p.Anchored = true
                        end
                    end
                end
                if antiGrabEnabled then
                    pcall(function() obj:Destroy() end)
                end
            end
        end
    end
    -- 🛡【アンチファイ】
    if antiPhiEnabled and myRoot then
        if myRoot.AssemblyLinearVelocity.Magnitude > 120 then
            myRoot.CFrame = lastSafeCFrame
            myRoot.AssemblyLinearVelocity = Vector3.zero
        else
            if tick() - lastUpdate > 0.3 then
                lastSafeCFrame = myRoot.CFrame
                lastUpdate = tick()
            end
        end
    end
    -- 🛡【アンチ爆弾】
    if antiBombEnabled and myRoot then
        if myRoot.AssemblyLinearVelocity.Magnitude > 45 and not antiPhiEnabled then
            myRoot.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            myRoot.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
        end
    end
    -- 🛡【アンチ火】
    if antiFireEnabled and myChar then
        for _, f in ipairs(myChar:GetDescendants()) do
            local nameLower = f.Name:lower()
            if f:IsA("Fire") or f:IsA("ParticleEmitter") or nameLower:find("fire") or nameLower:find("flame") or nameLower:find("burn") or nameLower:find("hit") then
                pcall(function()
                    if f:IsA("ParticleEmitter") then
                        f.Enabled = false
                    else
                        f:Destroy()
                    end
                end)
            end
        end
    end
    -- アンチキック
    if antiKickEnabled and myChar then
        local bpack = LocalPlayer:FindFirstChild("Backpack")
        local shuriken = bpack and (bpack:FindFirstChild("Shuriken") or bpack:FindFirstChild("手裏剣"))
        if shuriken then
            for _, v in ipairs(workspace:GetChildren()) do
                if v.Name == "CreatureBlobman" and (myRoot and (v.PrimaryPart and (v.PrimaryPart.Position - myRoot.Position).Magnitude < 25)) then
                    pcall(function()
                        shuriken.Parent = myChar
                        shuriken:Activate()
                    end)
                end
            end
        end
    end
end)

    local playerList = {}

    end,

-- ==========================================
-- 3. 🔒 選択したプレイヤーのみに送る（トグル）
-- ==========================================
LoggerTab:AddToggle({
    Name = "🔒 選択したプレイヤーのみロガー開始",
    CurrentValue = false,
    Callback = function(v)
        TargetLoggerRunning = v
        
        if v then
            -- プレイヤーが選択されているか確認
            if SelectedPlayerName == "" or not Players:FindFirstChild(SelectedPlayerName) then
                Rayfield:Notify({
                    Title = "エラー",
                    Content = "ターゲットが未選択か、既にサーバーにいません。",
                    Duration = 3
                })
                return
            end
            
            -- 非同期でループ処理を開始
            task.spawn(function()
                while TargetLoggerRunning do
                    local targetPlayer = Players:FindFirstChild(SelectedPlayerName)
                    
                    -- 対象が抜ける、またはトグルがオフになったらループ終了
                    if not targetPlayer or not TargetLoggerRunning then
                        break
                    end
                    
                    Rayfield:Notify({
                        Title = "個別ロガー稼働中",
                        Content = targetPlayer.Name .. " に送信中...",
                        Duration = LogSpeed -- スライダーの速度と同期
                    })
                    
                    -- スライダーで設定された秒数待機
                    task.wait(LogSpeed) 
                end
            end)
        end
    end
})
