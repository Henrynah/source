local function lookAt(startPosition, targetPosition)
    local directionVector = (targetPosition - startPosition).Unit
    local rightVector = directionVector:Cross(Vector3.new(0, 1, 0))
    local upVector = rightVector:Cross(directionVector)
    return CFrame.fromMatrix(startPosition, rightVector, upVector)
end

local function TeleportTo(cf)
    if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
        player.Character.HumanoidRootPart.CFrame = cf
        player.Character.HumanoidRootPart.Velocity = Vector3.new(0,0,0) 
        player.Character.HumanoidRootPart.AssemblyLinearVelocity = Vector3.new(0,0,0)
    end
end

local function lookPosition()
    if targetMode == "Player" and selectedPlayer then
        local targetChar = selectedPlayer.Character
        if targetChar and targetChar:FindFirstChild("HumanoidRootPart") then
            return targetChar.HumanoidRootPart.Position
        end
    end
    
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = { player.Character }
    params.FilterType = Enum.RaycastFilterType.Blacklist
    local result = workspace:Raycast(
        camera.CFrame.Position,
        camera.CFrame.LookVector * 1000,
        params
    )
    return result and result.Position or (camera.CFrame.Position + camera.CFrame.LookVector * 100)
end

local function makeDraggable(topbarObject, object)
    local dragging = false
    local dragInput, dragStart, startPos

    local function update(input)
        local delta = input.Position - dragStart
        local targetPos = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        TweenService:Create(object, TweenInfo.new(0.15), {Position = targetPos}):Play()
    end

    topbarObject.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = object.Position
            
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)

    topbarObject.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            update(input)
        end
    end)
end

local function prepareStuff()
    if preparingStuff then return end
    preparingStuff = true
    local MenuToys = ReplicatedStorage:FindFirstChild("MenuToys")
    if not MenuToys then
        notify("MenuToys not found", Color3.fromRGB(220, 60, 60))
        preparingStuff = false
        return
    end
    local SpawnToyRemoteFunction = MenuToys:FindFirstChild("SpawnToyRemoteFunction")
    if not SpawnToyRemoteFunction then
        notify("Spawn Remote not found", Color3.fromRGB(220, 60, 60))
        preparingStuff = false
        return
    end
    notify("Preparing..", Color3.fromRGB(100, 140, 220))
    
    -- Spawn items and track when each spawns
    local lastSpawnTime = tick()
    for i = 1, explosiveCount do
        local success = pcall(function()
            SpawnToyRemoteFunction:InvokeServer(currentMode, CFrame.new(-166, -5, -604), Vector3.new(0, -75.5, 0))
        end)
        task.wait(0.8)
    end
    
    local success = pcall(function()
        SpawnToyRemoteFunction:InvokeServer("Campfire", CFrame.new(-329, -5, -553), Vector3.new(0, 110.5, 0))
    end)
	
    task.wait(0.5)
    
    local spawnedItems = {}
    local folder = toysFolder()
    if not folder then
        notify("Toys folder not found", Color3.fromRGB(220, 60, 60))
        preparingStuff = false
        return
    end
    for _, item in ipairs(folder:GetChildren()) do
        if (item.Name == currentMode or item.Name == "Campfire") and item:IsA("Model") and item.PrimaryPart then
            table.insert(spawnedItems, item)
        end
    end
    if #spawnedItems == 0 then
        notify("Nothing to prepare", Color3.fromRGB(220, 60, 60))
        preparingStuff = false
        return
    end
    
    local originalCFrame = player.Character.HumanoidRootPart.CFrame
    for i, item in ipairs(spawnedItems) do
        if item and item.Parent then
            local grabPart = nil
            if item.Name == "Campfire" then grabPart = item:FindFirstChild("FirePlayerPart")
            elseif item.Name == "BombBalloon" then grabPart = item:FindFirstChild("Balloon")
            elseif item.Name == "BombMissile" then grabPart = item:FindFirstChild("Body") end
            if not grabPart then grabPart = item.PrimaryPart end
            if grabPart then
                TeleportTo(grabPart.CFrame * CFrame.new(5, 5, 0))
                task.wait(0.055) 
                pcall(function() CreateGrabLine:FireServer(grabPart, CFrame.new(0, 0, 0)) end)
                pcall(function()
                    local lookAtCF = lookAt(player.Character.HumanoidRootPart.Position, grabPart.Position)
                    SetNetworkOwner:FireServer(grabPart, lookAtCF)
                end)
                pcall(function() DestroyGrabLine:FireServer(grabPart) end)
                task.wait(0.055)
            end
        end
    end
    TeleportTo(originalCFrame)
    notify("Prepared " .. #spawnedItems .. " explosives", Color3.fromRGB(80, 180, 100))
    preparingStuff = false
end

local function activateCannon()

    local folder = toysFolder()
    if not folder then notify("No toys folder", Color3.fromRGB(220, 60, 60)) return end
    local campfire = getCampfire(folder)
    if not campfire then notify("Campfire missing", Color3.fromRGB(220, 60, 60)) return end
    local items = {}
    for _, v in ipairs(folder:GetChildren()) do
        if v.Name == currentMode and v:IsA("Model") and v.PrimaryPart then
            table.insert(items, v)
        end
    end
    if #items < 1 then notify("No " .. currentMode:lower() .. "s found", Color3.fromRGB(220, 60, 60)) return end
    
    local center = lookPosition() + Vector3.new(0, GRID_HEIGHT, 0)
    local count = 0
	    if isfile('Audio/'..song) then
        local audio = getcustomasset('Audio/'..song)
        local a = Instance.new("Sound", workspace)
        a.SoundId = audio
        a.Looped = looped
        a:Play()
        a.Volume = volume
    end
    task.wait(1)
    for i = 1, math.min(#items, explosiveCount) do
        local model = items[i]
        local gridSize = math.ceil(math.sqrt(explosiveCount))
        local row = math.floor((i - 1) / gridSize)
        local col = (i - 1) % gridSize
        local offset = Vector3.new((col - (gridSize-1)/2) * GRID_SPACING, 0, (row - (gridSize-1)/2) * GRID_SPACING)
        local cf = CFrame.new(center + offset)
        local targetPart = model.PrimaryPart
        if model.Name == "BombBalloon" and model:FindFirstChild("Balloon") then targetPart = model.Balloon
        elseif model.Name == "BombMissile" and model:FindFirstChild("Body") then targetPart = model.Body end

        if targetPart then
            pcall(function() SetNetworkOwner:FireServer(targetPart, cf) end)
            model:SetPrimaryPartCFrame(cf)
            count = count + 1
        end
    end
    
    -- Set up touch events after positioning
    if campfire and campfire.PrimaryPart then
        for _, model in ipairs(items) do
            if model and model.Parent then
                for _, part in ipairs(model:GetDescendants()) do
                    if part:IsA("BasePart") then
                        part.Touched:Connect(function(hit)
                            if hit:IsA("BasePart") and (hit.Anchored or hit.Material == Enum.Material.Grass) then
                                campfire:SetPrimaryPartCFrame(part.CFrame)
                            end
                        end)
                    end
                end
            end
        end
    end
    
    notify("hitman ran NukeShot()", Color3.fromRGB(255, 100, 100))
end

local function stabShot()
    local folder = toysFolder()
    if not folder then notify("No toys folder", Color3.fromRGB(220, 60, 60)) return end
    local campfire = getCampfire(folder)
    if not campfire then notify("Campfire missing", Color3.fromRGB(220, 60, 60)) return end
    local items = {}
    for _, v in ipairs(folder:GetChildren()) do
        if v.Name == currentMode and v:IsA("Model") and v.PrimaryPart then
            table.insert(items, v)
        end
    end
    if #items < 1 then notify("No " .. currentMode:lower() .. "s found", Color3.fromRGB(220, 60, 60)) return end
    
    local target = lookPosition()
	if isfile('Audio/'..song2) then
        local audio = getcustomasset('Audio/'..song2)
        local a = Instance.new("Sound", workspace)
        a.SoundId = audio
        a.Looped = looped
        a:Play()
        a.Volume = volume
    end
    -- Teleport all missiles to same spot
    for i = 1, math.min(#items, explosiveCount) do
        local model = items[i]
        local targetPart = model.PrimaryPart
        if model.Name == "BombBalloon" and model:FindFirstChild("Balloon") then targetPart = model.Balloon
        elseif model.Name == "BombMissile" and model:FindFirstChild("Body") then targetPart = model.Body end

        if targetPart then
            local cf = CFrame.new(target)
            pcall(function() SetNetworkOwner:FireServer(targetPart, cf) end)
            model:SetPrimaryPartCFrame(cf)
        end
    end
    
    -- Rapidly teleport campfire to each missile to ensure all explode
    task.spawn(function()
        task.wait(0.1)
        if campfire and campfire.PrimaryPart then
            for i = 1, math.min(#items, explosiveCount) do
                local model = items[i]
                if model and model.Parent and model.PrimaryPart then
                    campfire:SetPrimaryPartCFrame(model.PrimaryPart.CFrame)
                    task.wait(0.05)
                end
            end
        end
    end)
    
    notify("hitman ran StabShot()", Color3.fromRGB(255, 100, 100))
end
