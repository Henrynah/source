    spawnRemote = ReplicatedStorage.MenuToys.SpawnToyRemoteFunction

local cachedParts = {}
local spawnedBlobs = {}
local grabConn = nil

-- Step 1: Break joints of all other players and collect their dismembered parts
DismemberBtn.MouseButton1Click:Connect(function()
    Status.Text = "Status: Dismembering players..."
    cachedParts = {}
    
    if grabConn then
        grabConn:Disconnect()
        grabConn = nil
        GrabSpamBtn.BackgroundColor3 = Color3.fromRGB(160, 40, 40)
        GrabSpamBtn.Text = "3. EXECUTE BLOB GRABSPAM"
    end
    
    pcall(function()
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer then
                local char = player.Character
                if char then
                    local hum = char:FindFirstChildOfClass("Humanoid")
                    if hum then
                        hum.PlatformStand = true
                        hum.Health = 0
                    end
                    
                    char:BreakJoints()
                    
                    for _, part in ipairs(char:GetDescendants()) do
                        if part:IsA("BasePart") then
                            part.Anchored = false
                            part.CanCollide = true
                            part.Massless = false
                            table.insert(cachedParts, part)
                        end
                    end
                end
            end
        end
    end)
    
    Status.Text = "Status: Dismembered (" .. #cachedParts .. " parts ready)"
end)
