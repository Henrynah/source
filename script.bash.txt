-- Utility Functions
local function GetCharacter()
    return localPlayer.Character or localPlayer.CharacterAdded:Wait()
end

local function getLocalHumanoid()
    local char = localPlayer.Character
    return char and char:FindFirstChildOfClass("Humanoid")
end

local function getCharInfo()
    local char = localPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    return char, hrp, hum
end

local function getInventory()
    return Workspace:FindFirstChild(localPlayer.Name .. "SpawnedInToys")
end

local function SpawnToy(name, cframe, vector)
    SpawnToyRemoteFunction:InvokeServer(name, cframe, vector or Vector3.zero)
    local inv = getInventory()
    return inv and inv:WaitForChild(name, 5)
end

local function WaitForChild(Parent, Name, Time)
    return Parent:FindFirstChild(Name) or Parent:WaitForChild(Name, Time or 3)
end

local function SetOwnerGrab(part)
    if part then
        local ge = ReplicatedStorage:FindFirstChild("GrabEvents")
        if ge and ge:FindFirstChild("SetNetworkOwner") then
            ge.SetNetworkOwner:FireServer(part, part.CFrame)
        end
    end
end

local function IsInRange(pos1, pos2, range)
    return (pos1 - pos2).Magnitude <= range
end

-- Blobman Related
local function spawnBlobmanIfNeeded()
    local inv = getInventory()
    if not inv then return nil end

    local existing = inv:FindFirstChild(Config.ToyName)
    if existing then return existing end

    local char = GetCharacter()
    if char and char:FindFirstChild("Head") then
        pcall(function()
            SpawnToy(Config.ToyName, char.Head.CFrame)
        end)
    end

    return inv:WaitForChild(Config.ToyName, 5)
end

local function sitOnBlobman(blobman)
    if not blobman then return end
    local seat = blobman:FindFirstChildWhichIsA("VehicleSeat", true)
    local hum = getLocalHumanoid()
    if seat and hum and not seat.Occupant then
        task.wait(0.1)
        pcall(function() seat:Sit(hum) end)
    end
end

-- Target Check
local function CanAuraTarget(targetPlayer)
    if not targetPlayer or targetPlayer == localPlayer then return false end
    if not targetPlayer.Character then return false end
    local hum = targetPlayer.Character:FindFirstChildOfClass("Humanoid")
    return hum and hum.Health > 0
end

-- ===== Kill Aura =====
local function runKillAura()
    local hpToggle = false
    while Config.Running do
        task.wait(0.01)
        local _, hrp = getCharInfo()
        if not hrp then continue end

        local blobman = spawnBlobmanIfNeeded()
        if not blobman then continue end

        sitOnBlobman(blobman)

        local scriptFolder = blobman:FindFirstChild("BlobmanSeatAndOwnerScript")
        local detector = blobman:FindFirstChild("LeftDetector")
        local weld = detector and detector:FindFirstChild("LeftWeld")

        if scriptFolder and detector and weld then
            local CreatureGrab = scriptFolder:FindFirstChild("CreatureGrab")
            local CreatureRelease = scriptFolder:FindFirstChild("CreatureRelease")
            local CreatureDrop = scriptFolder:FindFirstChild("CreatureDrop")

            if CreatureGrab and CreatureRelease and CreatureDrop then
                hpToggle = not hpToggle
                for _, plr in ipairs(Players:GetPlayers()) do
                    if not Config.Running then break end

                    if plr ~= localPlayer then
                        local targetChar = plr.Character
                        local targetHRP = targetChar and targetChar:FindFirstChild("HumanoidRootPart")
                        local targetHumanoid = targetChar and targetChar:FindFirstChildOfClass("Humanoid")

                        if targetHRP and targetHumanoid and (hrp.Position - targetHRP.Position).Magnitude <= Config.Range then
                            pcall(function()
                                CreatureGrab:FireServer(detector, targetHRP, weld)
                                CreatureRelease:FireServer(weld, targetHRP)
                                CreatureDrop:FireServer(weld, targetHRP)
                                targetHumanoid.Health = hpToggle and 0 or 100
                            end)
                        end
                    end
                end
            end
        end
    end
end

-- ===== Launch Aura =====
function onLaunchAura()
    Config.LaunchAuraCoroutine = coroutine.create(function()
        while true do
            pcall(function()
                local char = GetCharacter()
                local root = char and char:FindFirstChild("HumanoidRootPart")
                if not root then return end

                for _, player in pairs(Players:GetPlayers()) do
                    if CanAuraTarget(player) then
                        local torso = player.Character:FindFirstChild("Torso") or player.Character:FindFirstChild("UpperTorso")
                        if torso and (torso.Position - root.Position).Magnitude <= Config.auraRadius then
                            SetNetworkOwner:FireServer(torso, player.Character.HumanoidRootPart.CFrame)
                            task.wait(0.1)
                            local bodyVel = torso:FindFirstChild("LaunchForce") or Instance.new("BodyVelocity", torso)
                            bodyVel.Name = "LaunchForce"
                            bodyVel.Velocity = Vector3.new(0, 200000000000, 0)
                            bodyVel.MaxForce = Vector3.new(0, math.huge, 0)
                            Debris:AddItem(bodyVel, 1)
                        end
                    end
                end
            end)
            task.wait(0.02)
        end
    end)
    coroutine.resume(Config.LaunchAuraCoroutine)
end

function offLaunchAura()
    if Config.LaunchAuraCoroutine then
        coroutine.close(Config.LaunchAuraCoroutine)
        Config.LaunchAuraCoroutine = nil
    end
end

-- ===== Walk Speed =====
local function onWalkSpeed()
    if Config.walkSpeedConnection then
        Config.walkSpeedConnection:Disconnect()
        Config.walkSpeedConnection = nil
    end

    Config.walkSpeedConnection = RunService.Heartbeat:Connect(function()
        if not Config.WalkSpeedToggle then return end
        local char = localPlayer.Character
        if char and char:FindFirstChild("Humanoid") and char:FindFirstChild("HumanoidRootPart") then
            local hrp = char.HumanoidRootPart
            local moveDir = char.Humanoid.MoveDirection
            if moveDir.Magnitude > 0 then
                hrp.CFrame = hrp.CFrame + moveDir * (Config.walkSpeedValue / 10)
            end
        end
    end)
end

local function offWalkSpeed()
    Config.WalkSpeedToggle = false
    if Config.walkSpeedConnection then
        Config.walkSpeedConnection:Disconnect()
        Config.walkSpeedConnection = nil
    end
    local char = localPlayer.Character
    if char and char:FindFirstChild("Humanoid") then
        char.Humanoid.WalkSpeed = 16
    end
end

-- ===== PVP Aura =====
local function PvpAuraLoop()
    local GE = ReplicatedStorage:FindFirstChild("GrabEvents")
    if not GE then return end
    local SetOwner = GE:FindFirstChild("SetNetworkOwner")

    while Config.PvpAuraActive do
        pcall(function()
            local char = localPlayer.Character
            if not char then task.wait(0.1) return end
            local root = char:FindFirstChild("HumanoidRootPart")
            if not root then task.wait(0.1) return end

            for _, player in pairs(Players:GetPlayers()) do
                if player ~= localPlayer then
                    local targetChar = player.Character
                    if targetChar then
                        local targetRoot = targetChar:FindFirstChild("HumanoidRootPart")
                        if targetRoot then
                            local dist = (root.Position - targetRoot.Position).Magnitude
                            if dist <= Config.PvpAuraRadius then
                                if Config.PvpAuraMode == "Grab" then
                                    SetOwner:FireServer(targetRoot, targetRoot.CFrame)
                                    local bp = targetRoot:FindFirstChild("AuraBP") or Instance.new("BodyPosition", targetRoot)
                                    bp.Name = "AuraBP"
                                    bp.MaxForce = Vector3.new(9e9, 9e9, 9e9)
                                    bp.P = 6000000
                                    bp.D = 500
                                    bp.Position = root.Position + Vector3.new(0, 5, 0)
                                elseif Config.PvpAuraMode == "Fling" then
                                    local bv = targetRoot:FindFirstChild("AuraBV") or Instance.new("BodyVelocity", targetRoot)
                                    bv.Name = "AuraBV"
                                    bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
                                    bv.Velocity = Vector3.new(
                                        math.random(-100000, 100000),
                                        math.random(50000, 200000),
                                        math.random(-100000, 100000)
                                    )
                                    task.wait(0.1)
                                    bv:Destroy()
                                elseif Config.PvpAuraMode == "Freeze" then
                                    for _, part in pairs(targetChar:GetChildren()) do
                                        if part:IsA("BasePart") then
                                            part.Anchored = true
                                        end
                                    end
                                    task.wait(0.5)
                                    for _, part in pairs(targetChar:GetChildren()) do
                                        if part:IsA("BasePart") then
                                            part.Anchored = false
                                        end
                                    end
                                elseif Config.PvpAuraMode == "Kill" then
                                    local hum = targetChar:FindFirstChild("Humanoid")
                                    if hum and hum.Health > 0 then
                                        hum.Health = 0
                                    end
                                elseif Config.PvpAuraMode == "Poison" then
                                    local hum = targetChar:FindFirstChild("Humanoid")
                                    if hum and hum.Health > 0 then
                                        hum.Health = hum.Health - 5
                                        task.wait(0.1)
                                    end
                                elseif Config.PvpAuraMode == "Ragdoll" then
                                    local RagdollRemote = ReplicatedStorage:FindFirstChild("CharacterEvents") and 
                                                        ReplicatedStorage.CharacterEvents:FindFirstChild("RagdollRemote")
                                    if RagdollRemote then
                                        RagdollRemote:FireServer(targetRoot, 0)
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end)
        task.wait(0.1)
    end
end

-- ===== Combo Attack =====
local function ComboAttackFunction()
    local GE = ReplicatedStorage:WaitForChild("GrabEvents")
    local SetOwner = GE:FindFirstChild("SetNetworkOwner")
    local RagdollRemote = ReplicatedStorage:FindFirstChild("CharacterEvents") and 
                         ReplicatedStorage.CharacterEvents:FindFirstChild("RagdollRemote")

    while Config.ComboActive do
        if GlobalTarget == "" then
            task.wait(0.5)
            continue
        end
        local target = Players:FindFirstChild(GlobalTarget)
        if not target then task.wait(0.5) continue end
        local targetChar = target.Character
        local targetRoot = targetChar and targetChar:FindFirstChild("HumanoidRootPart")
        local targetHumanoid = targetChar and targetChar:FindFirstChild("Humanoid")

        if targetRoot and targetHumanoid and targetHumanoid.Health > 0 then
            local myChar = localPlayer.Character
            local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
            if not myRoot then task.wait(0.1) continue end

            Config.ComboStep = Config.ComboStep + 1
            if Config.ComboStep > 6 then Config.ComboStep = 1 end

            if Config.ComboStep == 1 then
                SetOwner:FireServer(targetRoot, targetRoot.CFrame)
            elseif Config.ComboStep == 2 then
                if RagdollRemote then RagdollRemote:FireServer(targetRoot, 0) end
            elseif Config.ComboStep == 3 then
                myRoot.CFrame = targetRoot.CFrame * CFrame.new(0, 0, 3)
                local bp = targetRoot:FindFirstChild("ComboBP") or Instance.new("BodyPosition", targetRoot)
                bp.Name = "ComboBP"
                bp.MaxForce = Vector3.new(9e9, 9e9, 9e9)
                bp.P = 6000000
                bp.D = 500
                bp.Position = targetRoot.Position + Vector3.new(0, 30, 0)
            elseif Config.ComboStep == 4 then
                local bv = targetRoot:FindFirstChild("ComboBV") or Instance.new("BodyVelocity", targetRoot)
                bv.Name = "ComboBV"
                bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
                bv.Velocity = Vector3.new(math.random(-50000, 50000), 80000, math.random(-50000, 50000))
                task.wait(0.2)
                bv:Destroy()
            elseif Config.ComboStep == 5 then
                for _, part in pairs(targetChar:GetChildren()) do
                    if part:IsA("BasePart") then
                        part.Anchored = true
                    end
                end
                task.wait(0.5)
                for _, part in pairs(targetChar:GetChildren()) do
                    if part:IsA("BasePart") then
                        part.Anchored = false
                    end
                end
            elseif Config.ComboStep == 6 then
                targetHumanoid.Health = 0
            end
            task.wait(0.8)
        else
            task.wait(0.5)
        end
    end
end

                local partOwner = head:FindFirstChild("PartOwner")
                if partOwner then
                    local struggle = ReplicatedStorage:FindFirstChild("CharacterEvents") and ReplicatedStorage.CharacterEvents:FindFirstChild("Struggle")
                    if struggle then
                        struggle:FireServer()
                    end
                    for _, part in pairs(char:GetChildren()) do
                        if part:IsA("BasePart") then
                            part.Anchored = true
                        end
                    end
                    task.wait(0.1)
                    for _, part in pairs(char:GetChildren()) do
                        if part:IsA("BasePart") then
                            part.Anchored = false
                        end
                    end
                end
            end
        end)
        task.wait(0.05)
    end
end

-- ===== Anti Ragdoll =====
local function AntiRagdollLoop()
    while Config.AntiRagdollEnabled do
        pcall(function()
            local char = localPlayer.Character
            if char then
                local hum = char:FindFirstChild("Humanoid")
                if hum then
                    if hum.PlatformStand == true then
                        hum.PlatformStand = false
                    end
                    if hum.BreakJointsOnDeath then
                        hum.BreakJointsOnDeath = false
                    end
                    if hum.Health <= 0 then
                        hum.Health = 1
                    end
                    if hum.Sit and hum.SeatPart == nil then
                        hum.Sit = false
                    end
                end
            end
        end)
        task.wait(0.05)
    end
end

-- ===== Kill Bypass Loop =====
local function StartPlatformLoop()
    while Config.loopActive do
        local char = localPlayer.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        local camera = Workspace.CurrentCamera

        if root and camera then
            if not Config.platformPart and Config.spawnPlatformOnStart then
                Config.platformPart = Instance.new("Part", Workspace)
                Config.platformPart.Name = "SkyBase"
                Config.platformPart.Anchored = true
                Config.platformPart.Size = Vector3.new(1500, 2, 1500)
                Config.platformPart.CFrame = CFrame.new(0, Config.teleportHeight, 0)
                Config.platformPart.Transparency = 1
                Config.platformPart.CanCollide = false
                Workspace.FallenPartsDestroyHeight = -9999999
            end

            local originalPos = root.CFrame
            local savedCameraCFrame = camera.CFrame
            camera.CameraType = Enum.CameraType.Scriptable
            camera.CFrame = savedCameraCFrame

            root.CFrame = Config.platformPart.CFrame + Vector3.new(0, 5, 0)
            root.AssemblyLinearVelocity = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero

            task.wait(Config.loopInterval)

            if not Config.loopActive then
                char = localPlayer.Character
                root = char and char:FindFirstChild("HumanoidRootPart")
                if root then
                    root.CFrame = originalPos
                end
                camera.CameraType = Enum.CameraType.Custom
                break
            end

            char = localPlayer.Character
            root = char and char:FindFirstChild("HumanoidRootPart")
            if root then
                root.CFrame = originalPos
                root.AssemblyLinearVelocity = Vector3.zero
                root.AssemblyAngularVelocity = Vector3.zero
            end

            camera.CameraType = Enum.CameraType.Custom
            task.wait(Config.loopInterval)
        else
            task.wait(0.05)
        end
    end

    local camera = Workspace.CurrentCamera
    if camera then
        camera.CameraType = Enum.CameraType.Custom
    end
end
