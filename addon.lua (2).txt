local ncConn, ncDisabled = nil, false
local function enableNoclip()
    if not ncConn then
        ncDisabled = false
        ncConn = RunService.Stepped:Connect(function()
            if not ncDisabled and LP.Character then
                for _, v in ipairs(LP.Character:GetDescendants()) do
                    if v:IsA("BasePart") and v.CanCollide then v.CanCollide = false end
                end
            end
        end)
    end
end
local function disableNoclip()
    if ncConn then ncConn:Disconnect(); ncConn = nil end
    ncDisabled = true
end

-- ===== 所有権 =====
function lookAt(a, b)
    local d = (b - a).Unit; local r = d:Cross(Vector3.new(0, 1, 0))
    return CFrame.fromMatrix(a, r, r:Cross(d))
end
function SNOWship(p)
    if p and typeof(p) == "Instance" and LP.Character and getRoot() and LP:DistanceFromCharacter(p.Position) <= 30 then
        SetOwner:FireServer(p, lookAt(getRoot().Position, p.Position))
    end
end
function CheckNetOwner(plr)
    return plr and plr.Character and plr.Character:FindFirstChild("Head")
        and plr.Character.Head:FindFirstChild("PartOwner")
        and plr.Character.Head.PartOwner.Value == LP.Name
end
function GetPlayerCFrame()
    local c = getChar(); return c and c:FindFirstChild("HumanoidRootPart") and c.HumanoidRootPart.CFrame
end
function TeleportPlayer(cf)
    local c = getChar()
    if c and typeof(cf) == "CFrame" then c.HumanoidRootPart.CFrame = c.HumanoidRootPart.CFrame.Rotation + cf.Position end
end
function IsInSafeZone(plr) return plr:FindFirstChild("InPlot") and plr.InPlot.Value end
function IsFloating(plr) return plr.Character and getHum(plr.Character) and getHum(plr.Character).FloorMaterial == Enum.Material.Air end

        bv.Velocity = Vector3.new(0, 1e14, 0); bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    end
end

                for i, o in ipairs(drawObjects) do if o and o.Parent then local a = (i-1) / cnt * 2 * math.pi + drawAnimAngle; local idx = math.clamp(math.floor(((math.sin(a) + 1) / 2) * (#drawSavedPoints - 1)) + 1, 1, #drawSavedPoints); local pt = drawSavedPoints[idx]; local nx = (pt.X - minX) / rw; local ny = (pt.Y - minY) / rh; o.CFrame = CFrame.new(bp + Vector3.new((nx - 0.5) * drawSize * (fw / fh), (ny - 0.5) * drawSize, 0), bp) end end
            end
        end
    end
end)

    if espData[plr] then local d = espData[plr]; if d.Box then d.Box:Remove() end; if d.Name then d.Name:Remove() end; if d.Tracer then d.Tracer:Remove() end; espData[plr] = nil end
