local function notify(title, desc, time)
    Library:Notify({ Title = title or "FTP Hub", Description = desc or "", Time = time or 4 })
end

local function getInv()
    return Workspace:FindFirstChild(Player.Name .. "SpawnedInToys")
end

local function spawntoy(name, cf)
    if SpawnToy then
        pcall(function()
            SpawnToy:InvokeServer(name, cf or CFrame.new(HRP.Position + Vector3.new(0, 5, 0)), Vector3.new(0, 0, 0))
        end)
    end
end

local function destroytoy(toy)
    if DestroyToy and toy then
        pcall(function() DestroyToy:FireServer(toy) end)
    end
end

local function sno(obj)
    if SetNetworkOwner and obj then
        pcall(function() SetNetworkOwner:FireServer(obj, obj.CFrame) end)
    end
end

local function stopVelocity()
    if StopAllVelocity then
        pcall(function() StopAllVelocity:FireServer() end)
    end
end

local selectedPlayer = nil
local function getPlayerList()
    local list = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= Player then table.insert(list, p.Name) end
    end
    return list
end

-- Anti Grab (struggle loop)
local autoStruggleConn = nil
DefMain:AddToggle("AntiGrab", {
    Text    = "Anti Grab",
    Default = false,
    Tooltip = "Automatically struggles when grabbed",
    Callback = function(val)
        if val then
            autoStruggleConn = R.Heartbeat:Connect(function()
                local c = Player.Character
                if c and c:FindFirstChild("Head") then
                    local head = c.Head
                    if head:FindFirstChild("PartOwner") then
                        task.spawn(function()
                            pcall(function()
                                if StruggleEvent then StruggleEvent:FireServer(Player) end
                                stopVelocity()
                            end)
                            for _, p in pairs(c:GetChildren()) do
                                if p:IsA("BasePart") then p.Anchored = true end
                            end
                            local ih = Player:FindFirstChild("IsHeld")
                            while ih and ih.Value do task.wait() end
                            for _, p in pairs(c:GetChildren()) do
                                if p:IsA("BasePart") then p.Anchored = false end
                            end
                        end)
                    end
                end
            end)
        else
            if autoStruggleConn then autoStruggleConn:Disconnect(); autoStruggleConn = nil end
            local c = Player.Character
            if c then for _, p in pairs(c:GetChildren()) do if p:IsA("BasePart") then p.Anchored = false end end end
        end
    end
})

-- Anti Paint
local paintBackup = {}
local paintConns  = {}
DefExtra:AddToggle("AntiPaint", {
    Text    = "Anti Paint",
    Default = false,
    Callback = function(val)
        if val then
            for _, obj in ipairs(Workspace:GetDescendants()) do
                if obj:IsA("BasePart") and obj.Name == "PaintPlayerPart" then
                    local clone = obj:Clone(); clone.Archivable = true
                    paintBackup[obj:GetDebugId()] = { clone = clone, parent = obj.Parent }
                    obj:Destroy()
                end
            end
            table.insert(paintConns, Workspace.DescendantAdded:Connect(function(obj)
                if obj:IsA("BasePart") and obj.Name == "PaintPlayerPart" then
                    task.defer(function()
                        if obj and obj.Parent then
                            local clone = obj:Clone(); clone.Archivable = true
                            paintBackup[obj:GetDebugId()] = { clone = clone, parent = obj.Parent }
                            obj:Destroy()
                        end
                    end)
                end
            end))
        else
            for _, conn in ipairs(paintConns) do if conn.Connected then conn:Disconnect() end end
            paintConns = {}
            for _, data in pairs(paintBackup) do
                if data.clone and data.parent then data.clone.Parent = data.parent end
            end
            paintBackup = {}
        end
    end
})

    Func = function()
        local c   = Player.Character or Player.CharacterAdded:Wait()
        local hum = c:WaitForChild("Humanoid")
        local hrp = c:WaitForChild("HumanoidRootPart")
        safePos   = hrp.Position

        spawntoy("CreatureBlobman", CFrame.new(0, 5000000, 0))
        task.wait(1)

        local inv  = getInv()
        local blob = inv and inv:FindFirstChild("CreatureBlobman")
        local seat = blob and blob:FindFirstChildWhichIsA("VehicleSeat")

        if seat then
            hrp.CFrame = seat.CFrame + Vector3.new(0, 2, 0)
            seat:Sit(hum)
            notify("Anti Gucci", "Blobman seat active!", 3)
        end

        if antiGucciConn then antiGucciConn:Disconnect() end
        antiGucciConn = R.Heartbeat:Connect(function()
            if not hrp then return end
            if RagdollRemote then pcall(function() RagdollRemote:FireServer(hrp, 0) end) end
            if restoreFrames > 0 then
                hrp.CFrame = CFrame.new(safePos)
                restoreFrames -= 1
            end
        end)
    end
})

    Text = "Stop Anti Gucci",
    Func = function()
        if antiGucciConn then antiGucciConn:Disconnect(); antiGucciConn = nil end
        local inv  = getInv()
        local blob = inv and inv:FindFirstChild("CreatureBlobman")
        if blob then destroytoy(blob) end
        notify("Anti Gucci", "Stopped.", 2)
    end
})

local flyConn   = nil

local flyGyro   = nil
local flyActive = false
PlayerExtra:AddToggle("Fly", {
    Text    = "Fly",
    Default = false,
    Callback = function(val)
        flyActive = val
        local c = Player.Character
        if not c then return end
        local hrp = c:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        if val then
            local bp = Instance.new("BodyVelocity")
            bp.Velocity  = Vector3.zero
            bp.MaxForce  = Vector3.new(1e5, 1e5, 1e5)
            bp.Parent    = hrp
            local bg = Instance.new("BodyGyro")
            bg.MaxTorque = Vector3.new(1e5, 1e5, 1e5)
            bg.CFrame    = hrp.CFrame
            bg.Parent    = hrp
            flyBody = bp; flyGyro = bg
            flyConn = R.Heartbeat:Connect(function()
                local speed = 60
                local dir   = Vector3.zero
                local cf    = Camera.CFrame
                if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir += cf.LookVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir -= cf.LookVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir -= cf.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir += cf.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir += Vector3.new(0, 1, 0) end
                if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then dir -= Vector3.new(0, 1, 0) end
                bp.Velocity = dir.Magnitude > 0 and dir.Unit * speed or Vector3.zero
                bg.CFrame   = Camera.CFrame
            end)
        else
            if flyConn  then flyConn:Disconnect(); flyConn = nil end
            if flyBody  then flyBody:Destroy(); flyBody = nil end
            if flyGyro  then flyGyro:Destroy(); flyGyro = nil end
        end
    end
})

local FunMain  = Tabs.Fun:AddLeftGroupbox("Fun Tools")
local FunDecoy = Tabs.Fun:AddRightGroupbox("Decoy Control")

-- Decoy follow
local decoyConns = {}
local followMode = true

FunDecoy:AddToggle("DecoyFollowMode", {
    Text    = "Decoy Follow Mode (ON=follow / OFF=surround)",
    Default = true,
    Callback = function(val) followMode = val end
})

FunDecoy:AddButton({ Text = "Attach Decoy Control", Func = function()
    for _, c in ipairs(decoyConns) do c:Disconnect() end
    decoyConns = {}

    local decoys = {}
    for _, d in pairs(Workspace:GetDescendants()) do
        if d:IsA("Model") and d.Name == "YouDecoy" then table.insert(decoys, d) end
    end

    if #decoys == 0 then notify("Fun", "No decoys found!", 2); return end

    local mid = math.ceil(#decoys / 2)
    local conn = R.Heartbeat:Connect(function()
        for i, decoy in pairs(decoys) do
            local torso = decoy:FindFirstChild("Torso")
            if torso then
                local bp = torso:FindFirstChild("BodyPosition")
                local bg = torso:FindFirstChild("BodyGyro")
                if not bp then
                    bp = Instance.new("BodyPosition"); bp.MaxForce = Vector3.new(4e4, 4e4, 4e4); bp.D = 100; bp.P = 100; bp.Parent = torso
                end
                if not bg then
                    bg = Instance.new("BodyGyro"); bg.MaxTorque = Vector3.new(4e4, 4e4, 4e4); bg.D = 100; bg.P = 20000; bg.Parent = torso
                end
                if followMode then
                    local offset = (i - mid) * 5
                    local tgt    = HRP.Position - HRP.CFrame.LookVector * 5 + HRP.CFrame.RightVector * offset
                    bp.Position  = tgt
                    bg.CFrame    = CFrame.new(torso.Position, tgt)
                else
                    -- surround nearest player
                    local nearest, dist = nil, math.huge
                    for _, p in ipairs(Players:GetPlayers()) do
                        if p ~= Player and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                            local d = (HRP.Position - p.Character.HumanoidRootPart.Position).Magnitude
                            if d < dist then dist = d; nearest = p end
                        end
                    end
                    if nearest and nearest.Character then
                        local nRoot = nearest.Character.HumanoidRootPart
                        local angle = math.rad((i - 1) * (360 / #decoys))
                        bp.Position = nRoot.Position + Vector3.new(math.cos(angle) * 10, 0, math.sin(angle) * 10)
                        bg.CFrame   = CFrame.new(torso.Position, nRoot.Position)
                    end
                end
                sno(torso)
            end
        end
    end)
    table.insert(decoyConns, conn)
    notify("Fun", "Controlling " .. #decoys .. " decoys!", 3)
end })

FunDecoy:AddButton({ Text = "Disconnect Decoys", Func = function()
    for _, c in ipairs(decoyConns) do c:Disconnect() end
    decoyConns = {}
    notify("Fun", "Decoys disconnected.", 2)
end })
